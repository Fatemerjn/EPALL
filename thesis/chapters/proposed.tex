
\chapter{روش پیشنهادی}
\label{ch:proposed}

فصل پیشین نشان داد که نقاب هر وظیفه در دسترس است، اما \lr{PALL} از این نقاب‌ها برای تشخیص اهمیت وزن‌های مشترک استفاده نمی‌کند. منظور از درایه در روابط این فصل، یک مؤلفه از بردار پیراسنجه‌ها، یعنی یک وزن منفرد شبکه، است. ابتدا مسئله و سازوکار اصلی \lr{PALL} صورت‌بندی می‌شوند. سپس دلیل آسیب به وزن‌های مشترک بررسی و روش اصلی پایان‌نامه، \lr{EPALL}، همراه با الگوریتم و تحلیل پیچیدگی معرفی می‌شود. در ادامه، مسیر اکتشافی \lr{PALL-Adapter} و تحلیل نظری محدود آن ارائه می‌شوند. در پایان نیز ضوابط دسترسی به داده و محدودیت ادعاهای دو روش توضیح داده می‌شوند.

\medskip
\noindent\textbf{یادداشت نام‌گذاری.} روش اصلی این پایان‌نامه، نسخه‌ی توسعه‌یافته‌ی \lr{PALL} با نام \lr{EPALL} است (\lr{Enhanced PALL}). در کد پیاده‌سازی، گزارش‌های اجرا و برچسب‌های آزمایش، همین روش با شناسه‌ی \lr{pall\_modified} ثبت شده است. هر دو نام به یک الگوریتم اشاره می‌کنند.

\section{صورت‌بندی مسئله}
\label{sec:prop-formulation}

\subsection{نمادگذاری}
\label{subsec:prop-notation}

دنباله‌ی وظایف را با $\mathcal{T}=\{1,\ldots,n\}$ و مجموعه‌داده‌ی وظیفه‌ی $t$ را با $D_t$ نشان می‌دهیم؛ کلاس‌های وظایف مختلف مجزا هستند. شبکه با پیراسنجه‌های $\theta\in\mathbb{R}^{d}$ و یک سر طبقه‌بند به‌ازای هر وظیفه پیاده می‌شود. حافظه‌ی بازپخش $\mathcal{B}$ برای هر وظیفه‌ی $t$ سه‌گانه‌ی $(x,y,h)$ نگه می‌دارد که $h$ امتیازهای خام ثبت‌شده در پایان آموزش همان وظیفه است. سامانه دنباله‌ای از {درخواست‌ها\en{Requests}} را پردازش می‌کند؛ هر درخواست یا $\mathrm{learn}(t)$ است یا $\mathrm{forget}(t)$.

پس از دریافت $\mathrm{forget}(f)$، مجموعه‌ی وظایف فعال $\mathcal{T}_a=\{t\in\mathcal{T}: t \text{ آموزش‌دیده و حذف‌نشده}\}\setminus\{f\}$ است. دقت وظیفه‌ی $t$ در لحظه‌ی $r$ را با $A_t(r)$ و سطح تصادفی آن را با $c_t$ نشان می‌دهیم. زیان نگه‌داری روی حافظه‌ی وظایف فعال چنین است:
\begin{equation}
	\label{eq:retain-loss}
	\mathcal{L}_{\mathrm{retain}}(\theta)=\frac{1}{|\mathcal{T}_a|}\sum_{t\in\mathcal{T}_a}\ \mathbb{E}_{(x,y,h)\sim\mathcal{B}_t}\Big[\ell\big(f(x;\theta,t),y\big)\Big].
\end{equation}

\subsection{هدف مسئله}
\label{subsec:prop-objective}

هدف عملیاتی درخواست فراموشی، یافتن پیراسنجه‌های جدید $\theta'$ است که هم‌زمان دو شرط را برآورده کند:
\begin{equation}
	\label{eq:objective}
	\big|A_f(\theta')-c_f\big|\ \to\ 0
	\qquad\text{و}\qquad
	\max_{t\in\mathcal{T}_a}\big(A_t(\theta)-A_t(\theta')\big)\ \to\ 0 .
\end{equation}
شرط اول سرکوب هدف و شرط دوم حفظ بدترین وظیفه‌ی فعال است. عمدا از بیشینه و نه میانگین استفاده شده است: یک سامانه‌ی مستقر با میانگین خوب اما یک وظیفه‌ی ویران‌شده، از دید کاربران آن وظیفه شکست خورده است.

\subsection{سازوکار ارثی \lr{PALL}}
\label{subsec:prop-pall}

\lr{PALL} \cite{ozdenizci2025pall} برای هر وظیفه‌ی $t$ یک نقاب دودویی $M_t\in\{0,1\}^{d}$ می‌آموزد که زیرشبکه‌ی همان وظیفه را مشخص می‌کند؛ این نقاب با {امتیازهای قابل یادگیری\en{Learnable Scores}} و در سطح تنکی هدف به‌دست می‌آید و پس از آموزش ذخیره می‌شود \cite{kang2022wsn,ramanujan2020hidden}. استنتاج وظیفه‌ی $t$ با $f(x;\theta\odot M_t,t)$ انجام می‌گیرد.

در زمان درخواست فراموشی، \lr{PALL} سه کار انجام می‌دهد. نخست، نقاب $M_f$ را از مجموعه‌ی فعال کنار می‌گذارد و نمونه‌های بازپخش وظیفه‌ی هدف را حذف می‌کند. دوم، وزن‌های نقاب هدف را بر پایه‌ی وضعیت آن‌ها در نقاب‌های تاریخی و پشتیبان‌های ذخیره‌شده \emph{بازنشانی} یا \emph{بازسازی} می‌کند. سوم، یک مرحله‌ی ترمیم روی حافظه‌ی وظیفه‌های فعال اجرا می‌کند تا افت جانبی جبران شود.

نکته‌ای که در گام دوم مهم است و در فصل \ref{ch:results} به آن باز می‌گردیم این است که بازنشانی \emph{شاخه‌محور} است. اینکه یک درایه بازنشانی می‌شود یا حفظ می‌شود، به این بستگی دارد که وظیفه‌ی هدف پیش از وظایف فعال آموزش دیده باشد یا پس از آن‌ها، و اینکه پشتیبان تنک برای آن درایه ذخیره شده باشد یا نه. \lr{EPALL} این منطق شاخه‌محور را دست‌نخورده به ارث می‌برد؛ نوآوری آن جای دیگری است.

\section{محدودیت فراموشی بی‌قید}
\label{sec:prop-limitation}

فرض کنید در پاسخ به درخواست، به‌جای سازوکار ساختاریافته، یک گام بی‌قید در جهت افزایش زیان وظیفه‌ی هدف برداشته شود:
\begin{equation}
	\label{eq:naive-ascent}
	\theta' = \theta + \eta\,\nabla_\theta \mathcal{L}_{f}(\theta).
\end{equation}
اثر این گام بر یک وظیفه‌ی فعال $t$ را می‌توان با بسط مرتبه‌ی اول تقریب زد:
\begin{equation}
	\label{eq:collateral}
	\mathcal{L}_{t}(\theta')-\mathcal{L}_{t}(\theta)\ \approx\ \eta\,\big\langle \nabla_\theta\mathcal{L}_t(\theta),\ \nabla_\theta\mathcal{L}_f(\theta)\big\rangle .
\end{equation}
عبارت سمت راست، ضرب داخلی شیب وظیفه‌ی فعال و شیب وظیفه‌ی هدف است. اگر این دو شیب همسو باشند، آسیب جانبی مثبت و بزرگ است. اما ضرب داخلی یک مقدار \emph{انباشته} روی همه‌ی $d$ درایه است و همین انباشتگی، اطلاعات مهمی را پنهان می‌کند:
\begin{equation}
	\label{eq:decompose}
	\big\langle \nabla\mathcal{L}_t,\ \nabla\mathcal{L}_f\big\rangle
	= \underbrace{\sum_{i\in \Sshare}\partial_i\mathcal{L}_t\,\partial_i\mathcal{L}_f}_{\text{ناحیه‌ی مشترک}}
	+ \underbrace{\sum_{i\notin \Sshare}\partial_i\mathcal{L}_t\,\partial_i\mathcal{L}_f}_{\text{ناحیه‌ی غیرمشترک}} .
\end{equation}
در جمله‌ی دوم، درایه‌ها بیرون زیرشبکه‌ی وظایف فعال قرار دارند و بنابراین $|\partial_i\mathcal{L}_t|$ در آن‌ها کوچک است. پس \emph{انتظار می‌رود} سهم آن جمله کوچک باشد و بخش عمده‌ی آسیب جانبی از جمله‌ی نخست بیاید. تأکید می‌شود که این یک استدلال ساختاری است و نه اندازه‌گیری: سهم نسبی دو جمله در این کار به‌طور تجربی تفکیک نشده است. مهم‌تر آنکه درون همان جمله‌ی نخست هم توزیع یکنواخت نیست: چند درصد از درایه‌های مشترک، $|\partial_i\mathcal{L}_t|$ بزرگی دارند و بخش عمده‌ی مجموع را می‌سازند.

دو نتیجه‌ی طراحی از این تحلیل بیرون می‌آید. نتیجه‌ی اول، مهارکردن \emph{همه‌ی} به‌روزرسانی — آن‌گونه که یک تصویر سراسری در سبک \lr{GEM} \cite{lopezpaz2017gem} می‌کند — لازم نیست و هزینه‌ی آن کندشدن فراموشی است. نتیجه‌ی دوم، مهارکردن همه‌ی ناحیه‌ی مشترک هم لازم نیست، چون بخش بزرگی از آن کم‌اهمیت است. راه‌حل درست، مهارکردن \emph{حساس‌ترین} زیرمجموعه‌ی ناحیه‌ی مشترک است. این دقیقا کاری است که \lr{EPALL} می‌کند و بخش \ref{subsec:res-components} همین سه گزینه را به‌طور تجربی مقایسه می‌کند.

\subsection{هم‌پوشانی ساختاری}
\label{subsec:prop-overlap}

$\Mf$ نقاب دودویی ذخیره‌شده‌ی وظیفه‌ی درخواست‌شده برای حذف و $\Mr$ اجتماع نقاب‌های وظایف نگه‌داری‌شده است:
\begin{equation}
	\label{eq:union}
	\Mr=\bigvee_{t\in\mathcal{T}_a} M_t .
\end{equation}
{پشتیبان مشترک\en{Shared Support}} این دو، ناحیه‌ای است که هر تغییر در آن هم‌زمان به مسیر هدف و به حداقل یک مسیر فعال مربوط می‌شود:
\begin{equation}
	\label{eq:sshare}
	\Sshare=\Mf\cap\Mr .
\end{equation}
اهمیت این تعریف در دقیق‌بودن آن است. چون نقاب‌ها دودویی و ذخیره‌شده‌اند، $\Sshare$ یک اشتراک مجموعه‌ای است و به هیچ تقریب یا آستانه‌گذاری نیاز ندارد. این همان مزیتی است که ارث‌بری از خط زیرشبکه‌های نقاب‌دار \cite{kang2022wsn} فراهم می‌کند و در روش‌های بدون نقاب در دسترس نیست.

\subsection{رتبه‌بندی حساسیت و انتخاب بودجه}
\label{subsec:prop-ranking}

همه‌ی درایه‌های $\Sshare$ اهمیت یکسانی ندارند. برای هر $i\in\Sshare$ یک امتیاز حساسیت محاسبه می‌شود:
\begin{equation}
	\label{eq:sensitivity}
	q_i=\big|\nabla_i\mathcal{L}_{\mathrm{retain}}\big| ,
\end{equation}
که در آن شیب با یک گذر پس‌رو روی حافظه‌ی بازپخش وظایف فعال و روی زیرشبکه‌ی هر وظیفه محاسبه می‌شود. سپس با یک {نسبت محافظت\en{Protection Ratio}} $\rho\in(0,1]$، زیرمجموعه‌ی بحرانی انتخاب می‌شود:
\begin{equation}
	\label{eq:scrit}
	\Scrit=\Big\{ \text{$\lceil\rho\,|\Sshare|\rceil$ درایه‌ی برتر $\Sshare$ بر حسب $q_i$} \Big\}.
\end{equation}
مقدار پیش‌فرض $\rho=0.2$ است. توجیه این انتخاب در بخش \ref{sec:prop-limitation} آمد: سهم غالب آسیب جانبی از بخش بالایی توزیع $q_i$ می‌آید، پس یک بودجه‌ی کوچک کافی است و بودجه‌ی بزرگ‌تر بی‌جهت فراموشی را کند می‌کند.

یک جانشین برای معادله‌ی \eqref{eq:sensitivity} نیز پیاده‌سازی شده است: {انرژی تعارض شیب‌ها\en{Gradient-Conflict Energy}}
\begin{equation}
	\label{eq:conflict}
	q_i^{\mathrm{conf}}=\mathrm{ReLU}\!\big(-\,\nabla_i\mathcal{L}_{f}\cdot\nabla_i\mathcal{L}_{\mathrm{retain}}\big),
\end{equation}
که وزن‌هایی را هدف می‌گیرد که فراموشی و نگه‌داری در آن‌ها مستقیما با یکدیگر رقابت می‌کنند. این معیار به شیب وظیفه‌ی هدف نیاز دارد و بنابراین باید \emph{پیش از} حذف نمونه‌های بازپخش هدف محاسبه شود. این محدودیت، ترتیب گام‌های الگوریتم را تعیین می‌کند. بخش \ref{subsec:res-conflict} نشان می‌دهد نتیجه‌ی این معیار به مجموعه‌داده وابسته است؛ ازاین‌رو، بزرگی شیب معیار پیش‌فرض باقی می‌ماند.

\subsection{ترمیم لنگردار}
\label{subsec:prop-repair}

پس از اعمال بازنشانی شاخه‌محور ارثی، نقاب ترمیم اسمی به‌صورت $\Mf\cap\Mr$ ساخته می‌شود و حلقه‌ی ترمیم تنها روی همان درایه‌ها گام برمی‌دارد. تابع هدف ترمیم چنین است:
\begin{equation}
	\label{eq:repair}
	\mathcal{L}_{\mathrm{repair}}
	= \alpha\,\mathcal{L}_{DER}
	+ \beta\,\mathcal{L}_{CE}
	+ \lambda \sum_{i \in \Scrit} \big(\theta_i-\theta_i^{\mathrm{anchor}}\big)^2 .
\end{equation}
سه جمله سه نقش متفاوت دارند. $\mathcal{L}_{DER}$ امتیازهای خام ذخیره‌شده را تطبیق می‌دهد و بنابراین \emph{رفتار} پیشین مدل را بازمی‌گرداند \cite{buzzega2020dark}. $\mathcal{L}_{CE}$ از برچسب‌های وظایف نگه‌داری‌شده استفاده می‌کند و اطلاعات مستقیم طبقه‌بندی می‌دهد. جمله‌ی سوم، جمله‌ی {لنگر\en{Anchor}}، رانش را روی درایه‌های بحرانی انتخاب‌شده محدود می‌کند. ضریب پیش‌فرض $\lambda=1$ است.

از نظر شکل ریاضی، جمله‌ی سوم همان قالب \lr{EWC} \cite{kirkpatrick2017overcoming} است، اما دو تفاوت اساسی دارد که در بخش \ref{subsec:lit-reg} اشاره شد و اینجا صریح می‌شود: دامنه‌ی اعمال آن به $\Scrit$ محدود است و نه کل پیراسنجه‌ها، و هدف آن مهار یک عملیات \emph{حذف} است و نه یک عملیات \emph{یادگیری}.

\paragraph{انتخاب لنگر.} دو گزینه برای $\theta^{\mathrm{anchor}}$ وجود دارد. گزینه‌ی نخست $\theta^{\mathrm{old}}$ است، یعنی مقدار وزن بلافاصله پیش از درخواست؛ این طبیعی‌ترین انتخاب برای هدف نگه‌داری است اما یک ایراد مفهومی جدی دارد: $\theta^{\mathrm{old}}$ خودش در آموزش وظیفه‌ی فراموش‌شونده شکل گرفته و بنابراین اطلاعاتی از همان وظیفه حمل می‌کند. لنگرزدن یک عملیات \emph{حذف} به مرجعی که خودش وابسته به داده‌ی حذف‌شونده است، از نظر مفهومی خودشکن است. گزینه‌ی دوم $\theta^{\mathrm{reinit}}$ است، نمونه‌ای تازه از توزیع مقدارگذاری اولیه که به‌طور ساختاری از هر داده‌ای مستقل است.

این استدلال مفهومی به‌تنهایی $\theta^{\mathrm{reinit}}$ را پیشنهاد می‌کند، اما دو نکته این انتخاب را تضعیف می‌کنند. نخست، لنگر فقط روی $\Scrit \subseteq M_f \cap M_r$ عمل می‌کند، در حالی که فراموشی از راه بازنشانی ناحیه‌ی مخصوص هدف $M_f \setminus M_r$ رخ می‌دهد. بنابراین، لنگر برای حفظ وظیفه‌های دیگر به‌کار می‌رود و نه برای فراموشی؛ کشاندن وزن‌های مشترک به مقادیر تصادفی می‌تواند دانشی را که وظیفه‌های نگه‌داری‌شده در آن وزن‌ها دارند تخریب کند. دوم، $\theta^{\mathrm{reinit}}$ استقلال کامل از داده را ایجاد نمی‌کند، زیرا شاخه‌های بازنشانی همچنان از پشتیبان‌های تنک وابسته به داده‌های پیشین استفاده می‌کنند. مقایسه‌ی بخش \ref{subsec:res-anchor} با هشت بذر نیز نشان می‌دهد که پس از اصلاح هولم برای آزمون‌های متعدد، هیچ معیاری از نظر آماری معنادار نیست و نتایج \lr{CIFAR-100} از $\theta^{\mathrm{old}}$ پشتیبانی می‌کنند. بنابراین، $\theta^{\mathrm{old}}$ در پیاده‌سازی پیش‌فرض استفاده می‌شود و محدودیت مفهومی آن در بخش \ref{sec:concl-limitations} توضیح داده می‌شود.

\subsection{الگوریتم کامل}
\label{subsec:prop-algorithm}

الگوریتم \ref{algo:epall} کل فرایند پردازش درخواست را نشان می‌دهد. هر کمیت مبتنی بر شیب که به داده‌های وظیفه‌ی هدف نیاز دارد، پیش از حذف نمونه‌های بازپخش آن محاسبه می‌شود. پس از حذف، دیگر دسترسی به داده‌های هدف انجام نمی‌شود.

\begin{alg}{EPALL: request-time selective forgetting of task $f$}{epall}
\Require weights $\theta$, per-task masks $\{M_t\}$, rehearsal memory $\mathcal{B}$, active tasks $\mathcal{T}_a$, protection ratio $\rho$, anchor weight $\lambda$, repair steps $K$
\Ensure updated weights $\theta'$ with task $f$ suppressed
\State copy $M_f$; if conflict scoring is selected, cache its forget gradient while $\mathcal{B}_f$ is available
\State delete $\mathcal{B}_f$; remove $M_f$ from the active mask set \Comment{no target-data access beyond this point}
\State $M_r \gets \bigvee_{t \in \mathcal{T}_a} M_t$; \quad $S_{\mathrm{share}} \gets M_f \wedge M_r$
\State $q_i \gets |\nabla_i \mathcal{L}_{\mathrm{retain}}|$ for $i \in S_{\mathrm{share}}$ \Comment{one backward pass on $\mathcal{B}$}
\State $S_{\mathrm{crit}} \gets$ top $\lceil \rho\,|S_{\mathrm{share}}|\rceil$ coordinates of $S_{\mathrm{share}}$ by $q_i$
\State cache anchors $\theta^{\mathrm{anchor}}|_{S_{\mathrm{crit}}}$ and the sparse retained-task backups the inherited reset branches require
\State apply PALL's branch-dependent target reset/reconstruction
\State $M_{\mathrm{repair}} \gets M_f \wedge M_r$ \Comment{nominal retained-repair mask}
\For{$k = 1$ \textbf{to} $K$}
	\State sample retained rehearsal entries from $\mathcal{B}$
	\State $\mathcal{L} \gets \alpha \mathcal{L}_{DER} + \beta \mathcal{L}_{CE} + \lambda \sum_{i \in S_{\mathrm{crit}}} (\theta_i - \theta_i^{\mathrm{anchor}})^2$
	\State backward; zero gradients outside $M_{\mathrm{repair}}$; optimizer step
\EndFor
\State rebuild the active union mask; discard request-only target state
\State \textbf{return} $\theta$
\end{alg}

سه نکته‌ی پیاده‌سازی مهم است. نخست، گام $2$ یک محدودیت الزامی است. یک بررسی خودکار اطمینان می‌دهد که پس از حذف نمونه‌های بازپخش هدف، هیچ دسترسی دوباره‌ای به داده‌های آموزشی یا مجموعه‌ی نگه‌داری‌شده‌ی آن وظیفه انجام نشود. دوم، گام $6$ فقط پشتیبان‌های تنک موردنیاز بازنشانی را ذخیره می‌کند و این پشتیبان‌ها در محاسبه‌ی حافظه‌ی موردنیاز در بخش \ref{sec:res-storage} شمرده می‌شوند. سوم، پس از پایان درخواست، داده‌های موقت آن، از جمله لنگرها و نقاب بایگانی‌شده، کنار گذاشته می‌شوند؛ مگر آنکه اجرا برای ارزیابی برون‌خط انجام شده باشد.

\subsection{کنترل‌های مؤلفه‌ای}
\label{subsec:prop-controls}

برای اینکه ادعای سازوکاری قابل آزمون باشد، چهار کنترل هم‌تراز پیاده‌سازی شده است که هر یک \emph{تنها یک} جزء الگوریتم \ref{algo:epall} را عوض می‌کند و بقیه‌ی زمان‌بندی درخواست را دست‌نخورده می‌گذارد:

\begin{description}
	\item[بدون جریمه‌ی $\ell_2$.] $\lambda=0$. توجه شود که این کنترل \emph{محافظت را برنمی‌دارد}: ناحیه‌ی بحرانی همچنان شناسایی می‌شود و پشتیبان‌های تنک آن نیز ساخته می‌شوند؛ تنها جمله‌ی جریمه‌ی $\ell_2$ از تابع هدف حذف می‌شود. بنابراین نام درست آن «بدون جریمه‌ی $\ell_2$» است و نه «بدون محافظت»، و آنچه جدا می‌کند سهم \emph{شدت مهار} است و نه سهم کل سازوکار.
	\item[فقط هم‌پوشانی.] کل $\Sshare$ محافظت می‌شود، بدون هیچ رتبه‌بندی. این کنترل نشان می‌دهد آیا رتبه‌بندی حساسیت ارزشی افزون بر خود فیلتر ساختاری دارد.
	\item[بودجه‌ی تصادفی.] همان تعداد درایه‌ی کنترل کامل محافظت می‌شود، اما به‌طور تصادفی یکنواخت از میان $\Sshare$ انتخاب می‌شود. این کنترل، اثر \emph{اندازه‌ی} بودجه را از اثر \emph{کیفیت} انتخاب جدا می‌کند. توجه شود که نمونه‌گیری از $\Sshare$ انجام می‌شود و نه از $\Mf$؛ اگر از $\Mf$ نمونه گرفته می‌شد، درایه‌های منحصر به هدف در بودجه وارد می‌شدند که شیب‌شان بعدا در ترمیم نقاب می‌شود و در نتیجه بودجه‌ی \emph{مؤثر} کنترل از کنترل کامل کوچک‌تر می‌شد.
	\item[رتبه‌بندی بدون هم‌پوشانی.] رتبه‌بندی با همان $q_i$ روی کل زیرشبکه‌ی هدف انجام می‌شود، بدون اعمال صریح فیلتر $\Mr$. این کنترل می‌سنجد که آیا اشتراک صریح اطلاعاتی افزون بر رتبه‌بندی دارد.
\end{description}

نتایج این چهار کنترل در بخش \ref{subsec:res-components} گزارش می‌شود و — چنان‌که خواهیم دید — یکی از آن‌ها نتیجه‌ای می‌دهد که ساده‌سازی روش را پیشنهاد می‌کند.

\subsection{تحلیل پیچیدگی}
\label{subsec:prop-complexity}

هزینه‌ی افزوده‌ی \lr{EPALL} نسبت به \lr{PALL} اصلی سه جزء دارد. جزء اول، محاسبه‌ی $q_i$ است که یک گذر پس‌رو روی حافظه‌ی بازپخش وظایف فعال لازم دارد؛ هزینه‌ی آن $\mathcal{O}(|\mathcal{B}|\cdot C_{\mathrm{bwd}})$ است که $C_{\mathrm{bwd}}$ هزینه‌ی یک گذر پس‌رو است. چون $|\mathcal{B}|=500$ و ثابت است، این جزء با تعداد وظایف رشد نمی‌کند. جزء دوم، انتخاب $\lceil\rho|\Sshare|\rceil$ درایه‌ی برتر است که با انتخاب مرتبه‌ای در $\mathcal{O}(|\Sshare|)$ یا با مرتب‌سازی در $\mathcal{O}(|\Sshare|\log|\Sshare|)$ انجام می‌شود و در هر حال از $\mathcal{O}(d)$ فراتر نمی‌رود. جزء سوم، ارزیابی جمله‌ی لنگر در هر گام ترمیم است که $\mathcal{O}(|\Scrit|)$ و بنابراین حداکثر $\mathcal{O}(\rho\,d)$ است.

از نظر حافظه، \lr{EPALL} علاوه بر نقاب‌های \lr{PALL}، مقادیر لنگر روی $\Scrit$ را نگه می‌دارد که $\mathcal{O}(\rho|\Sshare|)$ است و پس از پایان درخواست آزاد می‌شود.

مجموع این هزینه‌ها از نظر نظری کوچک است، اما در عمل زمان پردازش درخواست را افزایش می‌دهد؛ زیرا یک گذر پس‌رو اضافه و جمله‌ی لنگر در هر یک از $K$ گام ترمیم محاسبه می‌شوند. بخش \ref{subsec:res-main} نشان می‌دهد \lr{EPALL} روی \lr{CIFAR-10} حدود $2.1$ ثانیه و روی \lr{CIFAR-100} حدود $12.1$ ثانیه از \lr{PALL} اصلی کندتر است. این نتیجه، موازنه‌ی میان حفظ بهتر وظیفه‌ها و افزایش زمان پردازش را نشان می‌دهد.

\section{مسیر اکتشافی \lr{PALL-Adapter}}
\label{sec:prop-adapter}

بخش‌های پیشین روش اصلی را کامل کردند. این بخش مسیر دومی را معرفی می‌کند که برای کاهش دامنه‌ی به‌روزرسانی طراحی شد، اما نتیجه‌ی نهایی آن منفی بود. این مسیر به دو دلیل گزارش می‌شود: طراحی و ارزیابی آن بخشی از پژوهش انجام‌شده است و پنج حالت آزمایشی هم‌تراز آن می‌توانند الگویی برای بررسی اجزای روش‌های مشابه باشند.

\subsection{معماری}
\label{subsec:prop-adapter-arch}

لایه‌های استخراج ویژگی $\theta_{\mathrm{base}}$ منجمد می‌شوند و تنها سه گروه پیراسنجه آموزش‌پذیر می‌مانند: یک {لایه‌ی تطبیق گلوگاهی مشترک\en{Shared Bottleneck Adapter}} با پیراسنجه‌های $\phi_{s}$، یک {لایه‌ی تطبیق مخصوص وظیفه\en{Task-Specific Adapter}} با پیراسنجه‌های $\phi_{t}$ به‌ازای هر وظیفه، و یک طبقه‌بند مشترک که از طریق برش سطری مخصوص وظیفه استفاده می‌شود. هر دو لایه‌ی تطبیق، بلوک‌های باقی‌مانده‌ی گلوگاهی استاندارد پس از بدنه با غیرخطیت \lr{ReLU} هستند \cite{houlsby2019parameter,nair2010relu} و به‌ترتیب اعمال می‌شوند: نخست مشترک، سپس مخصوص وظیفه. شکل \ref{fig:adapter-arch} این ساختار را نشان می‌دهد.

\subsection{فراموشی سه‌مرحله‌ای}
\label{subsec:prop-adapter-forget}

مراحل فراموشی در این مسیر ترتیب مشخصی دارند تا شیب‌های وابسته به هدف پیش از حذف نمونه‌های بازپخش آن محاسبه شوند.

\paragraph{مرحله یک: اندازه‌گیری هم‌پوشانی.} شیب‌های قدرمطلق لایه‌ی تطبیق مشترک روی حافظه‌ی هدف و حافظه‌ی نگه‌داری‌شده محاسبه می‌شوند و با نسبت‌های آستانه‌ای، دو مجموعه تعریف می‌شود: $\Sforget$ شامل درایه‌های با بالاترین شیب هدف، و $\Sactive$ شامل درایه‌های با بالاترین شیب نگه‌داری. اشتراک آن‌ها ناحیه‌ی بحرانی است:
\begin{equation}
	\label{eq:adapter-crit}
	\Scrit=\Sforget\cap\Sactive ,
\end{equation}
و $\lceil\alpha_p|\Scrit|\rceil$ درایه‌ی برتر آن بر حسب بزرگی شیب نگه‌داری، مجموعه‌ی {دارای حفاظت سخت\en{Hard-Protected}} $H$ را می‌سازد. توجه شود که این مجموعه‌ها با مجموعه‌های نقاب‌محور بخش \ref{sec:prop-epall} متفاوت‌اند: آن‌ها از نقاب‌های دودویی ذخیره‌شده می‌آیند و این‌ها از رتبه‌بندی شیب.

\paragraph{مرحله دو: بازنشانی هدف و به‌روزرسانی نقاب نرم.} لایه‌ی تطبیق مخصوص وظیفه‌ی هدف با همان قاعده‌ی مقدارگذاری اولیه بازنشانی می‌شود و برش طبقه‌بند آن پاک می‌شود. سپس لایه‌ی تطبیق مشترک در جهت رسیدن به توزیع یکنواخت روی کلاس‌های وظیفه‌ی هدف بهینه می‌شود، با این تفاوت که گام هر درایه در یک ضریب مقیاس ضرب می‌شود:
\begin{equation}
	\label{eq:softmask}
	m_{\mathrm{soft}}(i) = \begin{cases}
		0, & i \in H, \\
		1-p, & i \in \Scrit\setminus H, \\
		1, & i \in \Sforget\setminus\Sactive, \\
		0, & \text{در غیر این صورت,}
	\end{cases}
\end{equation}
که $p\in[0,1]$ شدت محافظت است. منطق این چهار حالت روشن است: ناحیه‌ی دارای حفاظت سخت تکان نمی‌خورد؛ باقی ناحیه‌ی بحرانی با ضریب کاهش‌یافته حرکت می‌کند؛ ناحیه‌ی منحصر به هدف آزادانه به‌روز می‌شود؛ و درایه‌های بیرون $\Sforget$ اصلا هدف این مرحله نیستند.

\paragraph{مرحله‌ی سه: صعود طبقه‌بند و ترمیم.} برش طبقه‌بند وظیفه‌ی هدف با استفاده از شیب فراموشی محاسبه‌شده \emph{پیش از بازنشانی}، یک گام صعود از روی شیب انجام می‌دهد. سپس نمونه‌های بازپخش هدف حذف می‌شوند و همه‌ی پیراسنجه‌های آموزش‌پذیر برای چند گام کوتاه روی حافظه‌ی وظیفه‌های نگه‌داری‌شده ترمیم می‌شوند.

\subsection{دامنه‌ی واقعی نقاب نرم}
\label{subsec:prop-adapter-scope}

نکته‌ی مهمی که پیش از هر تحلیل نظری باید تصریح شود، این است: ضریب $m_{\mathrm{soft}}$ تنها جابه‌جایی لایه‌ی تطبیق مشترک را \emph{در طول زیرمرحله هدف-یکنواخت} کنترل می‌کند. این ضریب یک نقاب محافظت دائمی نیست و دقت نگه‌داری آغاز تا پایان را کران نمی‌زند، چون سه عامل دیگر هم نقطه‌ی پایان را جابه‌جا می‌کنند: بازنشانی لایه‌ی تطبیق هدف، صعود مبتنی بر شیب طبقه‌بند، و ترمیم پایانی که نقاب‌نشده است. علاوه بر این، چون لایه‌ی تطبیق مشترک نهایی تابعی از شیب‌های حافظه‌ی هدف می‌ماند، این مسیر نه فراموشی دقیق است و نه گواهی‌شده از نظر حریم خصوصی.

\section{تحلیل نظری با دامنه‌ی محدود}
\label{sec:prop-theory}

این بخش یک گزاره‌ی دقیق درباره‌ی همان چیزی اثبات می‌کند که نقاب نرم واقعا کنترل می‌کند: تغییر مخاطره‌ی وظایف فعال \emph{در طول زیرمرحله نقاب نرم}. سپس بخش \ref{subsec:prop-theory-limits} توضیح می‌دهد چرا این گزاره به افت آغاز تا پایان تعمیم نمی‌یابد، و چرا این محدودیت با یافته‌ی تجربی فصل \ref{ch:results} سازگار است.

\subsection{نمادگذاری و افراز}
\label{subsec:prop-theory-notation}

در سراسر این تحلیل، بدنه‌ی منجمد $\theta_{\mathrm{base}}$ و لایه‌های تطبیق مخصوص وظیفه ثابت‌اند؛ پس تنها پیراسنجه‌های متغیر و در نتیجه تنها منبع تداخل میان‌وظیفه‌ای در این زیرمرحله، وزن‌های لایه‌ی تطبیق مشترک $\phi\triangleq\phi_s\in\mathbb{R}^{d_s}$ هستند. $\phi^{(0)}$ مقدار آن در آغاز زیرمرحله و $\phi^{(N)}$ مقدار آن پس از $N$ گام است. به‌روزرسانی هر درایه چنین است:
\begin{equation}
	\label{eq:coord-update}
	\phi^{(k+1)}_{i} = \phi^{(k)}_{i} - \eta\,m_{i}\,g^{(k)}_{i},
	\qquad
	g^{(k)}_{i} \triangleq \partial_{i}\,\mathcal{L}_{\mathrm{unlearn}}\big(\phi^{(k)}\big),
\end{equation}
که $m_i=m_{\mathrm{soft}}(i)\in[0,1]$ در طول $N$ گام ثابت است، چون نقاب یک‌بار در مرحله یک محاسبه می‌شود. مجموعه‌ی اندیس‌ها به چهار ناحیه‌ی مجزا افراز می‌شود:
\begin{equation}
	\label{eq:partition}
	\begin{aligned}
		\mathcal{F}^{\circ} &\triangleq \Sforget\setminus\Sactive && (m_i = 1),\\
		\mathcal{S} &\triangleq \Scrit\setminus H && (m_i = 1-p),\\
		H &\subseteq \Scrit && (m_i = 0),\\
		\mathcal{N} &\triangleq \{1,\dots,d_s\}\setminus \Sforget && (m_i = 0).
	\end{aligned}
\end{equation}

\begin{definition}[مخاطره‌ی نگه‌داری و افت زیرمرحله]
	\label{def:drop}
	برای هر وظیفه‌ی فعال $t\in\mathcal{T}_a$، مخاطره‌ی آن را به‌صورت تابعی از لایه‌ی تطبیق مشترک با $R_{t}(\phi)\triangleq\mathbb{E}_{(x,y)\sim D_t}[\ell(f(x;\phi),y)]$ تعریف می‌کنیم. افتی که \emph{مسیر زیرمرحله} $\phi^{(0)}\!\to\!\phi^{(N)}$ بر وظیفه‌ی $t$ تحمیل می‌کند برابر $\Delta_t\triangleq R_t(\phi^{(N)})-R_t(\phi^{(0)})$ است. تأکید می‌شود که $\Delta_t$ افت \emph{همین زیرمرحله} است و نه بیشترین افت گزارش‌شده در فصل \ref{ch:results}.
\end{definition}

\subsection{فرض‌ها}
\label{subsec:prop-theory-assumptions}

\begin{assumption}[همواری مخاطره‌ی نگه‌داری]
	\label{asm:smooth}
	هر $R_t$ روی پوشش محدب مسیر، $\beta$-هموار است: $\|\nabla R_t(\phi)-\nabla R_t(\phi')\|_2\leq\beta\|\phi-\phi'\|_2$.
\end{assumption}

\begin{assumption}[کران‌داری شیب‌های فراموشی]
	\label{asm:bounded}
	$|g^{(k)}_i|\leq G$ به‌طور یکنواخت روی مسیر؛ در نتیجه جابه‌جایی انباشته‌ی هر درایه از $\Gamma_i\triangleq|\sum_{k=0}^{N-1}g^{(k)}_i|\leq NG$ پیروی می‌کند.
\end{assumption}

\begin{assumption}[تمرکز حساسیت نگه‌داری روی ناحیه‌ی مشترک]
	\label{asm:concentration}
	$\epsilon\geq 0$ای وجود دارد که $|\partial_i R_t(\phi^{(0)})|\leq\epsilon$ برای هر $i\in\mathcal{F}^{\circ}$ و هر $t\in\mathcal{T}_a$.
\end{assumption}

فرض \ref{asm:concentration} یک \emph{فرض اضافی} درباره‌ی تک‌تک مخاطره‌های نگه‌داری است و از قاعده‌ی انتخاب الگوریتم به‌تنهایی نتیجه نمی‌شود. الگوریتم درایه‌ها را با بزرگی شیب \emph{تجمیعی} وظایف نگه‌داری رتبه‌بندی می‌کند؛ در نتیجه، خنثی‌شدن شیب‌های وظایف مختلف می‌تواند یک درایه با مشتق بزرگ برای یک وظیفه‌ی منفرد را بیرون مجموعه‌ی انتخابی نگه دارد. بنابراین لم بعدی مشروط به برقرار بودن این تمرکز است و اگر کوچک‌ترین $\epsilon$ معتبر بزرگ باشد، کران می‌تواند بی‌فایده شود.

\subsection{کران بالای افت زیرمرحله و یکنوایی آن نسبت به \texorpdfstring{$p$}{p}}
\label{subsec:prop-theory-prop}

\begin{lemma}[کران بالای افت زیرمرحله]
	\label{lem:subphase}
	فرض‌های \ref{asm:smooth} تا \ref{asm:concentration} را برقرار بگیرید و $\eta>0$ و $N$ را ثابت کنید. برای هر وظیفه‌ی فعال $t$، {انرژی تداخل بحرانی\en{Critical Interference Energy}} را چنین تعریف کنید:
	\begin{equation}
		\label{eq:Ecrit}
		E^{\mathrm{crit}}_{t} \triangleq \eta\!\!\sum_{i\in \Scrit}\!\!\big|\partial_i R_t(\phi^{(0)})\big|\,\Gamma_i ,
	\end{equation}
	و انرژی پس‌مانده‌ی منحصر به هدف را $C_t\triangleq\eta\sum_{i\in\mathcal{F}^{\circ}}\Gamma_i$. آنگاه به‌روزرسانی نقاب نرم با شدت محافظت $p$ در کران زیر صدق می‌کند:
	\begin{equation}
		\label{eq:soft-bound}
		\Delta_t^{\mathrm{soft}} \leq (1-p)\,E^{\mathrm{crit}}_t + \epsilon\,C_t + \tfrac{\beta}{2}\,\eta^2 N^2 G^2\Big(|\mathcal{F}^{\circ}| + (1-p)^2|\mathcal{S}|\Big),
	\end{equation}
	حال‌آنکه به‌روزرسانی بی‌قید ($m_i=1$ روی کل $\Sforget$) از کران زیر پیروی می‌کند:
	\begin{equation}
		\label{eq:unc-bound}
		\Delta_t^{\mathrm{unc}} \leq E^{\mathrm{crit}}_t + \epsilon\,C_t + \tfrac{\beta}{2}\,\eta^2 N^2 G^2\,|\Sforget| .
	\end{equation}
	سمت راست \eqref{eq:soft-bound} را \emph{کران جانشین} می‌نامیم و با $B_t(p)$ نشان می‌دهیم.
\end{lemma}

\begin{proof}
	وظیفه‌ی فعال $t$ را ثابت می‌گیریم و جابه‌جایی کل لایه‌ی تطبیق مشترک را $\delta\triangleq\phi^{(N)}-\phi^{(0)}$ می‌نامیم. با جمع‌بستن \eqref{eq:coord-update} روی $N$ گام و ثابت‌بودن $m_i$:
	\begin{equation}
		\label{eq:disp}
		\delta_i = -\eta\,m_i\sum_{k=0}^{N-1}g^{(k)}_i
		\qquad\Longrightarrow\qquad
		|\delta_i|\leq\eta\,m_i\,\Gamma_i\leq\eta\,m_i\,NG,
	\end{equation}
	که از نابرابری مثلثی و فرض \ref{asm:bounded} می‌آید.

	\textit{گام ۱ (بسط همواری).} طبق فرض \ref{asm:smooth}، $R_t$ کران بالای درجه‌دوم زیر را در نقطه‌ی پایه‌ی $\phi^{(0)}$ می‌پذیرد:
	\begin{equation}
		\label{eq:taylor}
		\Delta_t \leq \big\langle\nabla R_t(\phi^{(0)}),\,\delta\big\rangle + \tfrac{\beta}{2}\,\|\delta\|_2^2 .
	\end{equation}

	\textit{گام ۲ (جمله‌ی مرتبه‌ی اول).} با افراز \eqref{eq:partition} و نابرابری مثلثی:
	\begin{align}
		\big\langle\nabla R_t,\delta\big\rangle
		&\leq \sum_{i=1}^{d_s}\big|\partial_i R_t\big|\,|\delta_i| \notag\\
		&\leq \underbrace{\sum_{i\in\mathcal{F}^{\circ}}\!\big|\partial_i R_t\big|\,\eta\,\Gamma_i}_{m_i=1}
		+ \underbrace{\sum_{i\in\mathcal{S}}\!\big|\partial_i R_t\big|\,(1-p)\,\eta\,\Gamma_i}_{m_i=1-p}
		+ \underbrace{\sum_{i\in H\cup\mathcal{N}}\!\big|\partial_i R_t\big|\cdot 0}_{m_i=0}.
		\label{eq:three-sums}
	\end{align}
	جمله‌ی سوم به‌طور همانی صفر است. جمله‌ی نخست طبق فرض \ref{asm:concentration} حداکثر $\epsilon\,\eta\sum_{i\in\mathcal{F}^{\circ}}\Gamma_i=\epsilon C_t$ است. جمله‌ی دوم، چون $\mathcal{S}\subseteq\Scrit$ و همه‌ی جملاتش نامنفی‌اند، حداکثر $(1-p)E^{\mathrm{crit}}_t$ است. پس
	\begin{equation}
		\label{eq:first-order}
		\big\langle\nabla R_t,\delta\big\rangle \leq (1-p)\,E^{\mathrm{crit}}_t + \epsilon\,C_t .
	\end{equation}

	\textit{گام ۳ (جمله‌ی مرتبه‌ی دوم).} از \eqref{eq:disp}، $|\delta_i|^2\leq\eta^2 m_i^2 N^2 G^2$؛ و چون $m_i^2=1$ روی $\mathcal{F}^{\circ}$، $m_i^2=(1-p)^2$ روی $\mathcal{S}$، و $m_i=0$ در دو ناحیه‌ی دیگر:
	\begin{equation}
		\label{eq:second-order}
		\tfrac{\beta}{2}\|\delta\|_2^2 \leq \tfrac{\beta}{2}\,\eta^2 N^2 G^2\Big(|\mathcal{F}^{\circ}| + (1-p)^2|\mathcal{S}|\Big).
	\end{equation}
	جای‌گذاری \eqref{eq:first-order} و \eqref{eq:second-order} در \eqref{eq:taylor}، کران \eqref{eq:soft-bound} را می‌دهد.

	\textit{گام ۴ (خط پایه‌ی بی‌قید).} برای به‌روزرسانی بی‌قید، $m_i=1$ روی $\Sforget=\mathcal{F}^{\circ}\cup\Scrit$ و $m_i=0$ روی $\mathcal{N}$. تکرار گام‌های ۲ و ۳ با این ضرب‌گرها، جمله‌ی بحرانی را بدون مقیاس‌دهی و جمله‌ی انحنا را برابر $\tfrac{\beta}{2}\eta^2N^2G^2|\Sforget|$ می‌دهد؛ یعنی دقیقا \eqref{eq:unc-bound}.

\end{proof}

\begin{corollary}[یکنوایی کران جانشین نسبت به شدت محافظت]
	\label{cor:monotone}
	کران جانشین $B_t(p)=(1-p)E^{\mathrm{crit}}_t+\epsilon C_t+\tfrac{\beta}{2}\eta^2N^2G^2\big(|\mathcal{F}^{\circ}|+(1-p)^2|\mathcal{S}|\big)$ روی $p\in[0,1]$ مشتق‌پذیر و ناافزاینده است:
	\begin{equation}
		\label{eq:dBdp}
		\frac{\mathrm{d}B_t}{\mathrm{d}p} = -\,E^{\mathrm{crit}}_t \;-\; \beta\,\eta^2N^2G^2\,|\mathcal{S}|\,(1-p) \;\leq\; 0 .
	\end{equation}
	پس با بزرگ‌شدن $p$ کران کوچک‌تر می‌شود و نرخ کاهش آن دست‌کم $E^{\mathrm{crit}}_t$ است. در $p=1$ — یا هم‌ارز، با گسترش $H$ تا پوشش کل $\Scrit$ — سهم ناحیه‌ی بحرانی از کران کاملا حذف می‌شود و تنها $\epsilon C_t$ و جمله‌ی انحنای محدود به $\mathcal{F}^{\circ}$ باقی می‌ماند. افزون بر این $B_t(0)\leq B^{\mathrm{unc}}_t$، با تساوی دقیقا وقتی $H=\varnothing$.
\end{corollary}

\begin{observation}[آنچه از این کران \emph{نتیجه نمی‌شود}]
	\label{obs:no-contraction}
	از $\Delta^{\mathrm{soft}}_t\leq B_t(p)$ و $\Delta^{\mathrm{unc}}_t\leq B^{\mathrm{unc}}_t$ و $B_t(p)\leq B^{\mathrm{unc}}_t$ \emph{نمی‌توان} نتیجه گرفت که $\Delta^{\mathrm{soft}}_t\leq\Delta^{\mathrm{unc}}_t$ یا $\Delta^{\mathrm{soft}}_t\leq(1-p)\Delta^{\mathrm{unc}}_t$. مقایسه‌ی دو کران بالا هیچ رابطه‌ای میان دو مقدار واقعی برقرار نمی‌کند: هر دو مقدار می‌توانند به‌اندازه‌ی دلخواه از کران خود پایین‌تر باشند و ترتیبشان وارونه‌ی ترتیب کران‌ها شود. بنابراین ادعای این بخش تنها درباره‌ی \emph{کران بالای نظری} است و نه درباره‌ی افت واقعی.
\end{observation}

شکل \ref{fig:drop-decomposition} نشان می‌دهد که این کران جانشین با افزایش $p$ بزرگ‌تر نمی‌شود. با این حال، کران \eqref{eq:soft-bound} را نمی‌توان با گزارش‌های اجرا به‌طور مستقیم اعتبارسنجی کرد، زیرا طرف راست کران بر حسب مخاطره و افت اندازه‌گیری‌شده بر حسب دقت است و ثابت‌های $\epsilon$ و $\beta$ نیز واسنجی نشده‌اند. ازاین‌رو، حتی مشاهده‌ی همبستگی رتبه‌ای میان این دو کمیت نیز اعتبار کران را تأیید نمی‌کند؛ بنابراین هیچ ادعایی در این پایان‌نامه بر پایه‌ی چنین مقایسه‌ای بنا نشده است.

\subsection{چرا این کران، افت آغاز تا پایان را کران نمی‌زند}
\label{subsec:prop-theory-limits}

لم \ref{lem:subphase} درست است، اما دامنه‌ی آن باریک است و این محدودیت دامنه باید صریح بماند. سه شکاف میان $\Delta_t$ در تعریف \ref{def:drop} و بیشترین افت گزارش‌شده در فصل \ref{ch:results} وجود دارد:

\begin{enumerate}
	\item \textbf{بازنشانی پیش از زیرمرحله.} لایه‌ی تطبیق مخصوص وظیفه‌ی هدف و برش طبقه‌بند آن \emph{پیش از} شروع زیرمرحله نقاب نرم بازنشانی می‌شوند. این جابه‌جایی در $\phi^{(0)}$ گنجانده شده است، پس تحلیل درباره‌ی اثر آن هیچ نمی‌گوید — در حالی که همان اثر می‌تواند بخش عمده‌ی تغییر نقطه‌ی پایان باشد.
	\item \textbf{صعود طبقه‌بند.} گام صعود مبتنی بر شیب روی برش طبقه‌بند هدف، بیرون از فضای $\phi_s$ عمل می‌کند و نقاب نرم آن را نمی‌بیند.
	\item \textbf{ترمیم نقاب‌نشده.} مرحله سه همه‌ی پیراسنجه‌های آموزش‌پذیر — از جمله درایه‌های $H$ — را روی حافظه‌ی وظایف نگه‌داری‌شده به‌روز می‌کند. بنابراین حتی درایه‌هایی که در زیرمرحله قبل قفل بودند، در نقطه‌ی پایان جابه‌جا شده‌اند.
\end{enumerate}

نتیجه‌ی این سه شکاف آن است که یکنوایی کران جانشین نسبت به $p$ هیچ \emph{تضمینی} روی معیار گزارش‌شده نمی‌سازد. به همین دلیل، این پایان‌نامه لم \ref{lem:subphase} را به‌عنوان توضیح سازوکار زیرمرحله ارائه می‌کند و نه به‌عنوان پشتوانه‌ی نظری برای نتایج تجربی، و اثر واقعی نقاب نرم را با کنترل‌های مؤلفه‌ای می‌سنجد. بخش \ref{subsec:res-adapter-components} نشان می‌دهد آن کنترل‌ها هیچ سود قابل‌اندازه‌گیری برای نقاب نرم در نقطه‌ی پایان پیدا نمی‌کنند و سرکوب مشاهده‌شده را به بازنشانی مسیر هدف نسبت می‌دهند. این هم‌نشینی — یک کران درست با دامنه‌ی باریک، در کنار یک نتیجه‌ی تجربی منفی برای دامنه‌ی وسیع — تعارض نیست؛ دقیقا همان چیزی است که از تفکیک دقیق دامنه‌ها انتظار می‌رود.

\section{ضوابط دسترسی به داده}
\label{sec:prop-data-access}

هر دو مسیر پیشنهادی یک محدودیت مشترک دارند. همه‌ی کمیت‌های مبتنی بر شیب که به داده‌های وظیفه‌ی هدف نیاز دارند، پیش از حذف نمونه‌های بازپخش آن محاسبه و ذخیره می‌شوند. پس از این مرحله، دسترسی تازه‌ای به داده‌های هدف انجام نمی‌شود. بررسی‌های خودکار این محدودیت را در ابتدا و انتهای بازه‌ی دسترسی کنترل می‌کنند و در صورت نقض آن، اجرا متوقف می‌شود.

فقط نمونه‌های وظیفه‌ی هدف از حافظه‌ی بازپخش مدل حذف می‌شوند. داده‌های خام روی دیسک، نقاط وارسی، نسخه‌های پشتیبان و نسخه‌های بیرونی در دامنه‌ی این فرایند نیستند. بودجه‌ی بازپخش نیز دوباره توزیع نمی‌شود؛ سهم آزادشده‌ی وظیفه‌ی حذف‌شده به وظیفه‌های باقی‌مانده اختصاص نمی‌یابد تا افزایش ظرفیت حافظه با اثر روش فراموشی اشتباه گرفته نشود.

\section{دیدگاه یکپارچه و جمع‌بندی فصل}
\label{sec:prop-summary}

دو مسیر این فصل، دو شیوه‌ی اجرای یک ایده‌ی واحدند: \emph{ناحیه‌ی مشترک را بشناس، درون آن درایه‌های حساس به نگه‌داری را جدا کن، و تنها همان‌ها را مهار کن}. تفاوت‌ها در سه محور است. محور اول، تعریف ناحیه‌ی مشترک: در \lr{EPALL} این ناحیه اشتراک دقیق دو نقاب دودویی ذخیره‌شده است، در حالی که در \lr{PALL-Adapter} از اشتراک دو مجموعه‌ی رتبه‌بندی‌شده‌ی مبتنی بر شیب می‌آید و بنابراین به آستانه‌ها حساس است. محور دوم، شکل مهار: \lr{EPALL} یک جمله‌ی جریمه‌ی $\ell_2$ در تابع هدف ترمیم به‌کار می‌برد که در طول کل حلقه‌ی ترمیم فعال است، در حالی که \lr{PALL-Adapter} یک ضریب مقیاس روی گام به‌کار می‌برد که تنها در یک زیرمرحله فعال است. محور سوم، دامنه‌ی پیراسنجه‌ها: \lr{EPALL} روی کل ظرفیت شبکه عمل می‌کند و \lr{PALL-Adapter} روی پیمانه‌های کوچک در یک بدنه‌ی منجمد.

همان‌طور که فصل بعد نشان می‌دهد، دو تفاوت نخست تعیین‌کننده‌اند. محافظت پایدار از یک ناحیه‌ی دقیق مؤثر است، اما مقیاس‌دهی موقت روی یک ناحیه‌ی تقریبی نتیجه‌ی نهایی را تغییر نمی‌دهد. بر پایه‌ی این نتیجه، \lr{EPALL} روش اصلی پایان‌نامه است و \lr{PALL-Adapter} فقط به‌عنوان یک گزینه‌ی اکتشافی و ارزیابی‌شده گزارش می‌شود.
