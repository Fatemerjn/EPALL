
\chapter{روش پیشنهادی}
\label{ch:proposed}

فصل پیشین نشان داد که نقاب هر وظیفه در دسترس است، اما \lr{PALL} از این نقاب‌ها برای تشخیص اهمیت وزن‌های مشترک استفاده نمی‌کند. منظور از درایه در روابط این فصل، یک مؤلفه از بردار پیراسنجه‌ها، یعنی یک وزن منفرد شبکه، است. ابتدا مسئله و سازوکار اصلی \lr{PALL} صورت‌بندی می‌شوند. سپس دلیل آسیب به وزن‌های مشترک بررسی و روش اصلی پایان‌نامه، \lr{EPALL}، همراه با الگوریتم و تحلیل پیچیدگی معرفی می‌شود. در ادامه، مسیر اکتشافی \lr{PALL-Adapter} و تحلیل نظری محدود آن ارائه می‌شوند. در پایان نیز ضوابط دسترسی به داده و محدودیت ادعاهای دو روش توضیح داده می‌شوند.

\medskip
\noindent\textbf{یادداشت نام‌گذاری.} روش اصلی این پایان‌نامه، نسخه‌ی توسعه‌یافته‌ی \lr{PALL} با نام \lr{EPALL} است (\lr{Enhanced PALL}). در کد پیاده‌سازی، گزارش‌های اجرا و برچسب‌های آزمایش، همین روش با شناسه‌ی \lr{pall\_modified} ثبت شده است. هر دو نام به یک الگوریتم اشاره می‌کنند.

\section{صورت‌بندی مسئله}
\label{sec:prop-formulation}

\subsection{نمادگذاری}
\label{subsec:prop-notation}

دنباله‌ی وظایف را با $\mathcal{T}=\{1,\ldots,n\}$ و مجموعه‌داده‌ی وظیفه‌ی $t$ را با $D_t$ نشان می‌دهیم. کلاس‌های وظایف مختلف مجزا هستند. شبکه با پیراسنجه‌های $\theta\in\mathbb{R}^{d}$ و یک سر طبقه‌بند برای هر وظیفه پیاده می‌شود. حافظه‌ی بازپخش $\mathcal{B}$ برای هر وظیفه‌ی $t$ سه‌گانه‌ی $(x,y,h)$ را نگه می‌دارد که در آن $h$ امتیازهای خام ثبت‌شده در پایان آموزش همان وظیفه است. سامانه دنباله‌ای از درخواست‌ها را پردازش می‌کند. هر درخواست از نوع $\mathrm{learn}(t)$ یا $\mathrm{forget}(t)$ است.

پس از دریافت درخواست $\mathrm{forget}(f)$، مجموعه‌ی وظایف فعال از وظایفی تشکیل می‌شود که آموزش دیده‌اند، هنوز حذف نشده‌اند و با وظیفه‌ی $f$ تفاوت دارند. این مجموعه به‌صورت زیر تعریف می‌شود.
\[
	\mathcal{T}_a=\bigl\{t\in\mathcal{T}\mid t \text{ آموزش‌دیده و حذف‌نشده است}\bigr\}\setminus\{f\}.
\]
دقت وظیفه‌ی $t$ در مرحله‌ی $r$ را با $A_t(r)$ و دقت حدس تصادفی آن را با $c_t$ نشان می‌دهیم. زیان نگه‌داری روی حافظه‌ی وظایف فعال از رابطه‌ی زیر به دست می‌آید.
\begin{equation}
	\label{eq:retain-loss}
	\mathcal{L}_{\mathrm{retain}}(\theta)=\frac{1}{|\mathcal{T}_a|}\sum_{t\in\mathcal{T}_a}\ \mathbb{E}_{(x,y,h)\sim\mathcal{B}_t}\Big[\ell\big(f(x;\theta,t),y\big)\Big].
\end{equation}

\subsection{هدف مسئله}
\label{subsec:prop-objective}

هدف درخواست فراموشی، یافتن پیراسنجه‌های جدید $\theta'$ است که هم‌زمان دو شرط زیر را برآورده کند.
\begin{equation}
	\label{eq:objective}
	\big|A_f(\theta')-c_f\big|\ \to\ 0
	\qquad\text{و}\qquad
	\max_{t\in\mathcal{T}_a}\big(A_t(\theta)-A_t(\theta')\big)\ \to\ 0 .
\end{equation}
شرط اول به سرکوب وظیفه‌ی هدف و شرط دوم به حفظ عملکرد همه‌ی وظیفه‌های فعال مربوط است. در شرط دوم، بیشترین افت عملکرد در میان وظیفه‌های فعال معیار ارزیابی است و میانگین افت آن‌ها در نظر گرفته نمی‌شود. ممکن است میانگین عملکرد یک سامانه مناسب باشد، اما افت شدید یک وظیفه همچنان برای کاربران همان وظیفه به معنای شکست سامانه است.

\subsection{سازوکار ارثی \lr{PALL}}
\label{subsec:prop-pall}

\lr{PALL} \cite{ozdenizci2025pall} برای هر وظیفه‌ی $t$ یک نقاب دودویی $M_t\in\{0,1\}^{d}$ می‌آموزد که زیرشبکه‌ی همان وظیفه را مشخص می‌کند. این نقاب با {امتیازهای قابل یادگیری\en{Learnable Scores}} و در سطح تنکی هدف به‌دست می‌آید و پس از آموزش ذخیره می‌شود \cite{kang2022wsn,ramanujan2020hidden}. استنتاج وظیفه‌ی $t$ با $f(x;\theta\odot M_t,t)$ انجام می‌گیرد.

در زمان درخواست فراموشی، \lr{PALL} سه کار انجام می‌دهد. نخست، نقاب $M_f$ را از مجموعه‌ی فعال کنار می‌گذارد و نمونه‌های بازپخش وظیفه‌ی هدف را حذف می‌کند. دوم، وزن‌های نقاب هدف را بر پایه‌ی وضعیت آن‌ها در نقاب‌های تاریخی و پشتیبان‌های ذخیره‌شده \emph{بازنشانی} یا \emph{بازسازی} می‌کند. سوم، یک مرحله‌ی ترمیم روی حافظه‌ی وظیفه‌های فعال اجرا می‌کند تا افت جانبی جبران شود.

در گام دوم، قاعده‌ی بازنشانی برای همه‌ی درایه‌ها یکسان نیست. تصمیم به بازنشانی یا حفظ هر درایه به ترتیب آموزش وظیفه‌ی هدف و وظیفه‌های فعال و نیز وجود پشتیبان تنک برای آن درایه بستگی دارد. \lr{EPALL} این قاعده‌ی مبتنی بر وضعیت هر درایه را بدون تغییر از \lr{PALL} می‌گیرد و نوآوری آن به بخش دیگری از روش مربوط است. در فصل \ref{ch:results} دوباره به اثر این قاعده باز می‌گردیم.

\section{محدودیت فراموشی بی‌قید}
\label{sec:prop-limitation}

برای بررسی فراموشی بی‌قید، حالتی را در نظر می‌گیریم که در پاسخ به درخواست، به‌جای سازوکار ساختاریافته، یک گام در جهت افزایش زیان وظیفه‌ی هدف برداشته می‌شود.
\begin{equation}
	\label{eq:naive-ascent}
	\theta' = \theta + \eta\,\nabla_\theta \mathcal{L}_{f}(\theta).
\end{equation}
اثر این گام بر یک وظیفه‌ی فعال $t$ را می‌توان با بسط مرتبه‌ی اول تقریب زد.
\begin{equation}
	\label{eq:collateral}
	\mathcal{L}_{t}(\theta')-\mathcal{L}_{t}(\theta)\ \approx\ \eta\,\big\langle \nabla_\theta\mathcal{L}_t(\theta),\ \nabla_\theta\mathcal{L}_f(\theta)\big\rangle .
\end{equation}
عبارت سمت راست، ضرب داخلی شیب وظیفه‌ی فعال و شیب وظیفه‌ی هدف است. اگر این دو شیب همسو باشند، آسیب جانبی مثبت و بزرگ است. بااین‌حال، ضرب داخلی مقدار انباشته‌شده روی همه‌ی $d$ درایه‌ها را نشان می‌دهد و تفاوت میان ناحیه‌های مختلف را پنهان می‌کند.
\begin{equation}
	\label{eq:decompose}
	\big\langle \nabla\mathcal{L}_t,\ \nabla\mathcal{L}_f\big\rangle
	= \underbrace{\sum_{i\in \Sshare}\partial_i\mathcal{L}_t\,\partial_i\mathcal{L}_f}_{\text{ناحیه‌ی مشترک}}
	+ \underbrace{\sum_{i\notin \Sshare}\partial_i\mathcal{L}_t\,\partial_i\mathcal{L}_f}_{\text{ناحیه‌ی غیرمشترک}} .
\end{equation}
در جمله‌ی دوم، درایه‌ها بیرون زیرشبکه‌ی وظایف فعال قرار دارند و بنابراین انتظار می‌رود $|\partial_i\mathcal{L}_t|$ و سهم این جمله کوچک باشد. در نتیجه، احتمالا بخش عمده‌ی آسیب جانبی از جمله‌ی نخست می‌آید. این نتیجه تنها یک استدلال ساختاری است، زیرا سهم نسبی دو جمله در این پژوهش به‌طور تجربی جداگانه اندازه‌گیری نشده است. همچنین سهم درایه‌های جمله‌ی نخست یکسان نیست و بخش کوچکی از درایه‌های مشترک با مقدار بزرگ $|\partial_i\mathcal{L}_t|$ می‌توانند بخش عمده‌ی مجموع را تشکیل دهند.

از این تحلیل دو نکته برای طراحی روش به دست می‌آید. نخست، محدود کردن همه‌ی به‌روزرسانی‌ها، مانند محدودیت سراسری روش \lr{GEM} \cite{lopezpaz2017gem}، ضروری نیست و فراموشی را کند می‌کند. دوم، لازم نیست همه‌ی ناحیه‌ی مشترک محدود شود، زیرا بخش بزرگی از آن اهمیت اندکی دارد. بنابراین \lr{EPALL} تنها حساس‌ترین زیرمجموعه‌ی ناحیه‌ی مشترک را محدود می‌کند. بخش \ref{subsec:res-components} این راهکار را با دو حالت دیگر مقایسه می‌کند.

\subsection{هم‌پوشانی ساختاری}
\label{subsec:prop-overlap}

$\Mf$ نقاب دودویی ذخیره‌شده‌ی وظیفه‌ی درخواست‌شده برای حذف است و $\Mr$ اجتماع نقاب‌های وظیفه‌های نگه‌داری‌شده را نشان می‌دهد.
\begin{equation}
	\label{eq:union}
	\Mr=\bigvee_{t\in\mathcal{T}_a} M_t .
\end{equation}
{پشتیبان مشترک\en{Shared Support}} این دو نقاب، ناحیه‌ای است که هر تغییر در آن هم‌زمان بر مسیر هدف و دست‌کم یک مسیر فعال اثر می‌گذارد.
\begin{equation}
	\label{eq:sshare}
	\Sshare=\Mf\cap\Mr .
\end{equation}
این تعریف، ناحیه‌ی مشترک را به‌طور دقیق مشخص می‌کند. چون نقاب‌ها دودویی و ذخیره‌شده‌اند، $\Sshare$ یک اشتراک مجموعه‌ای است و به تقریب یا تعیین آستانه نیاز ندارد. این ویژگی از به‌کارگیری زیرشبکه‌های نقاب‌دار \cite{kang2022wsn} به دست می‌آید و در روش‌های بدون نقاب در دسترس نیست.

\subsection{رتبه‌بندی حساسیت و تعیین تعداد درایه‌های محافظت‌شده}
\label{subsec:prop-ranking}

همه‌ی درایه‌های $\Sshare$ اهمیت یکسانی ندارند. برای هر $i\in\Sshare$ امتیاز حساسیت زیر محاسبه می‌شود.
\begin{equation}
	\label{eq:sensitivity}
	q_i=\big|\nabla_i\mathcal{L}_{\mathrm{retain}}\big| ,
\end{equation}
در این رابطه، شیب با یک گذر پس‌رو روی حافظه‌ی بازپخش وظیفه‌های فعال و زیرشبکه‌ی هر وظیفه محاسبه می‌شود. سپس زیرمجموعه‌ی بحرانی با استفاده از {نسبت محافظت\en{Protection Ratio}} $\rho\in(0,1]$ انتخاب می‌شود.
\begin{equation}
	\label{eq:scrit}
	\Scrit=\Big\{ \text{$\lceil\rho\,|\Sshare|\rceil$ درایه‌ی برتر $\Sshare$ بر حسب $q_i$} \Big\}.
\end{equation}
مقدار پیش‌فرض $\rho=0.2$ است. تحلیل بخش \ref{sec:prop-limitation} نشان داد که بخش بالایی توزیع $q_i$ احتمالا بیشترین سهم را در آسیب جانبی دارد. بنابراین محافظت از تعداد کمی از درایه‌های دارای بیشترین امتیاز کافی است و افزایش بی‌دلیل این تعداد، فراموشی را کند می‌کند.

معیار دیگری نیز به‌عنوان جایگزین معادله‌ی \eqref{eq:sensitivity} پیاده‌سازی شده است که {انرژی تعارض شیب‌ها\en{Gradient-Conflict Energy}} نام دارد.
\begin{equation}
	\label{eq:conflict}
	q_i^{\mathrm{conf}}=\mathrm{ReLU}\!\big(-\,\nabla_i\mathcal{L}_{f}\cdot\nabla_i\mathcal{L}_{\mathrm{retain}}\big),
\end{equation}
این معیار وزن‌هایی را انتخاب می‌کند که فراموشی و نگه‌داری در آن‌ها مستقیما با یکدیگر رقابت می‌کنند. محاسبه‌ی آن به شیب وظیفه‌ی هدف نیاز دارد و باید پیش از حذف نمونه‌های بازپخش هدف انجام شود. این محدودیت، ترتیب گام‌های الگوریتم را تعیین می‌کند. بخش \ref{subsec:res-conflict} نشان می‌دهد که نتیجه‌ی این معیار به مجموعه‌داده وابسته است. ازاین‌رو، بزرگی شیب معیار پیش‌فرض باقی می‌ماند.

\subsection{ترمیم لنگردار}
\label{subsec:prop-repair}

پس از اجرای قاعده‌ی بازنشانی \lr{PALL}، نقاب مرحله‌ی ترمیم برابر $\Mf\cap\Mr$ در نظر گرفته می‌شود و حلقه‌ی ترمیم تنها درایه‌های همین ناحیه را به‌روز می‌کند. تابع هدف ترمیم به‌صورت زیر است.
\begin{equation}
	\label{eq:repair}
	\mathcal{L}_{\mathrm{repair}}
	= \alpha\,\mathcal{L}_{DER}
	+ \beta\,\mathcal{L}_{CE}
	+ \lambda \sum_{i \in \Scrit} \big(\theta_i-\theta_i^{\mathrm{anchor}}\big)^2 .
\end{equation}
سه جمله‌ی این تابع نقش‌های متفاوتی دارند. $\mathcal{L}_{DER}$ امتیازهای خام ذخیره‌شده را تطبیق می‌دهد و از این راه رفتار پیشین مدل را بازمی‌گرداند \cite{buzzega2020dark}. $\mathcal{L}_{CE}$ با استفاده از برچسب‌های وظیفه‌های نگه‌داری‌شده، اطلاعات مستقیم طبقه‌بندی را فراهم می‌کند. جمله‌ی سوم یا جمله‌ی {لنگر\en{Anchor}}، {رانش\en{Drift}} وزن‌های بحرانی انتخاب‌شده را محدود می‌کند. مقدار پیش‌فرض ضریب آن $\lambda=1$ است.

از نظر ریاضی، جمله‌ی سوم همان قالب \lr{EWC} \cite{kirkpatrick2017overcoming} را دارد، اما دو تفاوت اساسی میان آن‌ها وجود دارد. این جمله فقط روی $\Scrit$ اعمال می‌شود، نه روی همه‌ی پیراسنجه‌ها. همچنین هدف آن محدود کردن تغییرات هنگام حذف است، نه هنگام یادگیری. این تفاوت‌ها پیش‌تر در بخش \ref{subsec:lit-reg} نیز مطرح شدند.

\paragraph{انتخاب لنگر.} برای $\theta^{\mathrm{anchor}}$ دو گزینه وجود دارد. گزینه‌ی نخست، مقدار وزن درست پیش از دریافت درخواست است که با $\theta^{\mathrm{old}}$ نشان داده می‌شود. این گزینه برای حفظ وظیفه‌های فعال طبیعی به نظر می‌رسد، اما در آموزش وظیفه‌ی فراموش‌شونده شکل گرفته است و از آن وظیفه اطلاعات دارد. بنابراین وابسته کردن عملیات حذف به چنین مرجعی از نظر مفهومی ناسازگار است. گزینه‌ی دوم، نمونه‌ای تازه از توزیع مقدارگذاری اولیه و مستقل از داده‌هاست که با $\theta^{\mathrm{reinit}}$ نشان داده می‌شود.

این استدلال به‌تنهایی $\theta^{\mathrm{reinit}}$ را پیشنهاد می‌کند، اما دو نکته آن را تضعیف می‌کنند. نخست، لنگر فقط روی $\Scrit \subseteq M_f \cap M_r$ عمل می‌کند، در حالی که فراموشی با بازنشانی ناحیه‌ی مخصوص هدف $M_f \setminus M_r$ انجام می‌شود. پس نقش لنگر حفظ وظیفه‌های دیگر است، نه اجرای فراموشی. نزدیک کردن وزن‌های مشترک به مقادیر تصادفی ممکن است دانش وظیفه‌های نگه‌داری‌شده را تخریب کند. دوم، $\theta^{\mathrm{reinit}}$ استقلال کامل از داده را ایجاد نمی‌کند، زیرا قاعده‌ی بازنشانی همچنان از پشتیبان‌های تنک وابسته به داده‌های پیشین استفاده می‌کند. مقایسه‌ی بخش \ref{subsec:res-anchor} با هشت بذر نیز نشان می‌دهد که پس از اصلاح هولم برای آزمون‌های متعدد، هیچ معیاری از نظر آماری معنادار نیست و نتایج \lr{CIFAR-100} از $\theta^{\mathrm{old}}$ پشتیبانی می‌کنند. بنابراین، $\theta^{\mathrm{old}}$ در پیاده‌سازی پیش‌فرض به‌کار می‌رود و محدودیت مفهومی آن در بخش \ref{sec:concl-limitations} توضیح داده می‌شود.

\subsection{الگوریتم کامل}
\label{subsec:prop-algorithm}

الگوریتم \ref{algo:epall} کل فرایند پردازش درخواست را نشان می‌دهد. هر کمیت مبتنی بر شیب که به داده‌های وظیفه‌ی هدف نیاز دارد، پیش از حذف نمونه‌های بازپخش آن محاسبه می‌شود. پس از حذف، دیگر دسترسی به داده‌های هدف انجام نمی‌شود.

\begin{alg}{EPALL for request-time selective forgetting of task $f$}{epall}
\Require weights $\theta$, per-task masks $\{M_t\}$, rehearsal memory $\mathcal{B}$, active tasks $\mathcal{T}_a$, protection ratio $\rho$, anchor weight $\lambda$, repair steps $K$
\Ensure updated weights $\theta'$ with task $f$ suppressed
\State copy $M_f$; if conflict scoring is selected, cache its forget gradient while $\mathcal{B}_f$ is available
\State delete $\mathcal{B}_f$; remove $M_f$ from the active mask set \Comment{no target-data access beyond this point}
\State $M_r \gets \bigvee_{t \in \mathcal{T}_a} M_t$; \quad $S_{\mathrm{share}} \gets M_f \wedge M_r$
\State $q_i \gets |\nabla_i \mathcal{L}_{\mathrm{retain}}|$ for $i \in S_{\mathrm{share}}$ \Comment{one backward pass on $\mathcal{B}$}
\State $S_{\mathrm{crit}} \gets$ top $\lceil \rho\,|S_{\mathrm{share}}|\rceil$ coordinates of $S_{\mathrm{share}}$ by $q_i$
\State cache anchors $\theta^{\mathrm{anchor}}|_{S_{\mathrm{crit}}}$ and the sparse retained-task backups the inherited reset branches require
\State apply PALL's branch-dependent target reset/reconstruction
\State $M_{\mathrm{repair}} \gets M_f \wedge M_r$ \Comment{nominal retained-repair mask}
\For{$k = 1$ \textbf{to} $K$}
	\State sample retained rehearsal entries from $\mathcal{B}$
	\State $\mathcal{L} \gets \alpha \mathcal{L}_{DER} + \beta \mathcal{L}_{CE} + \lambda \sum_{i \in S_{\mathrm{crit}}} (\theta_i - \theta_i^{\mathrm{anchor}})^2$
	\State backward; zero gradients outside $M_{\mathrm{repair}}$; optimizer step
\EndFor
\State rebuild the active union mask; discard request-only target state
\State \textbf{return} $\theta$
\end{alg}

در پیاده‌سازی سه نکته اهمیت دارد. نخست، گام $2$ یک محدودیت الزامی است و بررسی خودکار تضمین می‌کند که پس از حذف نمونه‌های بازپخش هدف، دسترسی دوباره‌ای به داده‌های آموزشی یا مجموعه‌ی نگه‌داری‌شده‌ی آن وظیفه انجام نشود. دوم، گام $6$ فقط پشتیبان‌های تنک موردنیاز بازنشانی را ذخیره می‌کند. این پشتیبان‌ها در محاسبه‌ی حافظه‌ی موردنیاز در بخش \ref{sec:res-storage} شمرده می‌شوند. سوم، پس از پایان درخواست، داده‌های موقت آن از جمله لنگرها و نقاب بایگانی‌شده کنار گذاشته می‌شوند، مگر آنکه اجرا برای ارزیابی برون‌خط انجام شده باشد.

\subsection{کنترل‌های مؤلفه‌ای}
\label{subsec:prop-controls}

برای سنجش نقش هر جزء، چهار حالت مقایسه‌ای هم‌تراز پیاده‌سازی شده‌اند. هر حالت فقط یک جزء الگوریتم \ref{algo:epall} را تغییر می‌دهد و بقیه‌ی مراحل پردازش درخواست را ثابت نگه می‌دارد.

\begin{description}
	\item[بدون جریمه‌ی $\ell_2$.] در این حالت $\lambda=0$ است. ناحیه‌ی بحرانی همچنان شناسایی و پشتیبان‌های تنک آن ساخته می‌شوند و فقط جمله‌ی جریمه‌ی $\ell_2$ از تابع هدف حذف می‌شود. بنابراین این حالت اثر شدت محافظت را می‌سنجد و به معنای حذف کامل محافظت نیست.
	\item[فقط هم‌پوشانی.] همه‌ی درایه‌های $\Sshare$ بدون رتبه‌بندی محافظت می‌شوند. این حالت نشان می‌دهد که آیا رتبه‌بندی حساسیت، افزون بر فیلتر ساختاری، سودی دارد یا نه.
	\item[انتخاب تصادفی درایه‌ها.] تعداد درایه‌های محافظت‌شده با روش کامل یکسان است، اما درایه‌ها به‌طور تصادفی و یکنواخت از $\Sshare$ انتخاب می‌شوند. این حالت اثر تعداد درایه‌ها را از اثر کیفیت انتخاب آن‌ها جدا می‌کند. درایه‌ها فقط از مجموعه‌ی $\Sshare$ نمونه‌گیری می‌شوند و $\Mf$ مبنای نمونه‌گیری نیست. در صورت نمونه‌گیری از $\Mf$، ممکن است درایه‌های مخصوص هدف نیز انتخاب شوند. شیب این درایه‌ها هنگام ترمیم صفر می‌شود و در نتیجه تعداد درایه‌های واقعا محافظت‌شده از روش کامل کمتر خواهد بود.
	\item[رتبه‌بندی بدون هم‌پوشانی.] رتبه‌بندی با همان $q_i$ روی کل زیرشبکه‌ی هدف انجام می‌شود و فیلتر $\Mr$ اعمال نمی‌شود. این حالت مشخص می‌کند که آیا اشتراک صریح نقاب‌ها اطلاعاتی افزون بر رتبه‌بندی فراهم می‌کند یا نه.
\end{description}

نتایج این چهار حالت در بخش \ref{subsec:res-components} گزارش می‌شوند. نتیجه‌ی یکی از آن‌ها امکان ساده‌سازی روش را نشان می‌دهد.

\subsection{تحلیل پیچیدگی}
\label{subsec:prop-complexity}

هزینه‌ی افزوده‌ی \lr{EPALL} نسبت به \lr{PALL} اصلی سه جزء دارد. جزء اول، محاسبه‌ی $q_i$ است که به یک گذر پس‌رو روی حافظه‌ی بازپخش وظایف فعال نیاز دارد. هزینه‌ی آن $\mathcal{O}(|\mathcal{B}|\cdot C_{\mathrm{bwd}})$ است که در آن $C_{\mathrm{bwd}}$ هزینه‌ی یک گذر پس‌رو را نشان می‌دهد. چون $|\mathcal{B}|=500$ و ثابت است، این جزء با تعداد وظایف رشد نمی‌کند. جزء دوم، انتخاب $\lceil\rho|\Sshare|\rceil$ درایه‌ی برتر است. این انتخاب در $\mathcal{O}(|\Sshare|)$ یا با مرتب‌سازی در $\mathcal{O}(|\Sshare|\log|\Sshare|)$ انجام می‌شود و در هر حال از $\mathcal{O}(d)$ فراتر نمی‌رود. جزء سوم، ارزیابی جمله‌ی لنگر در هر گام ترمیم است که هزینه‌ای برابر $\mathcal{O}(|\Scrit|)$ و حداکثر $\mathcal{O}(\rho\,d)$ دارد.

از نظر حافظه، \lr{EPALL} علاوه بر نقاب‌های \lr{PALL}، مقادیر لنگر روی $\Scrit$ را نگه می‌دارد که $\mathcal{O}(\rho|\Sshare|)$ است و پس از پایان درخواست آزاد می‌شود.

مجموع این هزینه‌ها از نظر نظری کوچک است، اما در عمل زمان پردازش درخواست را افزایش می‌دهد، زیرا یک گذر پس‌رو اضافه و جمله‌ی لنگر در هر یک از $K$ گام ترمیم محاسبه می‌شوند. بخش \ref{subsec:res-main} نشان می‌دهد \lr{EPALL} روی \lr{CIFAR-10} حدود $2.1$ ثانیه و روی \lr{CIFAR-100} حدود $12.1$ ثانیه از \lr{PALL} اصلی کندتر است. این نتیجه، موازنه‌ی میان حفظ بهتر وظیفه‌ها و افزایش زمان پردازش را نشان می‌دهد.

\section{مسیر اکتشافی \lr{PALL-Adapter}}
\label{sec:prop-adapter}

بخش‌های پیشین روش اصلی را کامل کردند. این بخش مسیر دومی را معرفی می‌کند که برای کاهش دامنه‌ی به‌روزرسانی طراحی شد، اما نتیجه‌ی نهایی آن منفی بود. این مسیر گزارش می‌شود، زیرا طراحی و ارزیابی آن بخشی از پژوهش انجام‌شده است و پنج حالت آزمایشی هم‌تراز آن می‌توانند الگویی برای بررسی اجزای روش‌های مشابه باشند.

\subsection{معماری}
\label{subsec:prop-adapter-arch}

لایه‌های استخراج ویژگی $\theta_{\mathrm{base}}$ منجمد می‌شوند و تنها سه گروه پیراسنجه آموزش‌پذیر باقی می‌مانند. گروه نخست، یک {لایه‌ی تطبیق گلوگاهی مشترک\en{Shared Bottleneck Adapter}} با پیراسنجه‌های $\phi_{s}$ است. گروه دوم، یک {لایه‌ی تطبیق مخصوص وظیفه\en{Task-Specific Adapter}} با پیراسنجه‌های $\phi_{t}$ برای هر وظیفه است. گروه سوم نیز یک طبقه‌بند مشترک است که از برش سطری مخصوص هر وظیفه استفاده می‌کند. هر دو لایه‌ی تطبیق، بلوک‌های باقی‌مانده‌ی گلوگاهی استانداردی هستند که پس از بدنه قرار می‌گیرند و از غیرخطی بودن تابع فعال‌ساز \lr{ReLU} استفاده می‌کنند \cite{houlsby2019parameter,nair2010relu}. ابتدا لایه‌ی مشترک و سپس لایه‌ی مخصوص وظیفه اعمال می‌شود. شکل \ref{fig:adapter-arch} این ساختار را نشان می‌دهد.

مقدارگذاری اولیه‌ی لایه‌های تطبیق به‌گونه‌ای انجام می‌شود که {پیمانه\en{Module}} در آغاز مانند تابع همانی عمل کند. وزن‌های نگاشت کاهش ابعاد از توزیع کایمینگ \cite{he2015delving} نمونه‌گیری می‌شوند و وزن‌های نگاشت افزایش ابعاد با صفر مقدارگذاری می‌شوند. بنابراین افزودن لایه‌ی تطبیق به شبکه‌ی آموزش‌دیده، رفتار اولیه‌ی آن را تغییر نمی‌دهد.

\subsection{فراموشی سه‌مرحله‌ای}
\label{subsec:prop-adapter-forget}

مراحل فراموشی در این مسیر ترتیب مشخصی دارند تا شیب‌های وابسته به هدف پیش از حذف نمونه‌های بازپخش آن محاسبه شوند.

\paragraph{مرحله‌ی یک، اندازه‌گیری هم‌پوشانی.} قدرمطلق شیب‌های لایه‌ی تطبیق مشترک روی حافظه‌ی هدف و حافظه‌ی وظیفه‌های نگه‌داری‌شده محاسبه می‌شود. سپس با تعیین دو نسبت آستانه، مجموعه‌ی $\Sforget$ از درایه‌های دارای بیشترین شیب هدف و مجموعه‌ی $\Sactive$ از درایه‌های دارای بیشترین شیب نگه‌داری تشکیل می‌شود. اشتراک این دو مجموعه، ناحیه‌ی بحرانی را مشخص می‌کند.
\begin{equation}
	\label{eq:adapter-crit}
	\Scrit=\Sforget\cap\Sactive ,
\end{equation}
از میان درایه‌های $\Scrit$، تعداد $\lceil\alpha_p|\Scrit|\rceil$ درایه با بیشترین شیب نگه‌داری انتخاب می‌شود و مجموعه‌ی {دارای حفاظت کامل\en{Hard-Protected}} $H$ را می‌سازد. این مجموعه‌ها با مجموعه‌های نقاب‌محور بخش \ref{sec:prop-epall} تفاوت دارند. مجموعه‌های آن بخش از نقاب‌های دودویی ذخیره‌شده به دست می‌آیند، اما مجموعه‌های این بخش حاصل رتبه‌بندی شیب‌ها هستند.

\paragraph{مرحله‌ی دو، بازنشانی هدف و به‌روزرسانی نقاب نرم.} لایه‌ی تطبیق مخصوص وظیفه‌ی هدف با همان قاعده‌ی مقدارگذاری اولیه بازنشانی می‌شود و برش طبقه‌بند آن پاک می‌شود. سپس لایه‌ی تطبیق مشترک برای نزدیک کردن خروجی به توزیع یکنواخت روی کلاس‌های وظیفه‌ی هدف بهینه می‌شود. در این بهینه‌سازی، اندازه‌ی گام هر درایه در ضریب زیر ضرب می‌شود.
\begin{equation}
	\label{eq:softmask}
	m_{\mathrm{soft}}(i) = \begin{cases}
		0, & i \in H, \\
		1-p, & i \in \Scrit\setminus H, \\
		1, & i \in \Sforget\setminus\Sactive, \\
		0, & \text{در غیر این صورت,}
	\end{cases}
\end{equation}
در این رابطه $p\in[0,1]$ شدت محافظت را نشان می‌دهد. درایه‌های دارای حفاظت کامل تغییر نمی‌کنند. باقی ناحیه‌ی بحرانی با گامی کوچک‌تر و ناحیه‌ی مخصوص هدف بدون کاهش اندازه‌ی گام به‌روز می‌شود. درایه‌های بیرون $\Sforget$ نیز در این مرحله تغییر نمی‌کنند.

\paragraph{مرحله‌ی سه، صعود طبقه‌بند و ترمیم.} برش طبقه‌بند وظیفه‌ی هدف با استفاده از شیب فراموشی که پیش از بازنشانی محاسبه شده است، یک گام در جهت افزایش زیان برمی‌دارد. سپس نمونه‌های بازپخش هدف حذف می‌شوند و همه‌ی پیراسنجه‌های آموزش‌پذیر با چند گام کوتاه روی حافظه‌ی وظیفه‌های نگه‌داری‌شده ترمیم می‌شوند.

\subsection{دامنه‌ی واقعی نقاب نرم}
\label{subsec:prop-adapter-scope}

ضریب $m_{\mathrm{soft}}$ فقط تغییرات لایه‌ی تطبیق مشترک را در زیرمرحله‌ی یکنواخت‌سازی خروجی هدف محدود می‌کند. این ضریب یک نقاب محافظت دائمی نیست و برای افت عملکرد از آغاز تا پایان درخواست کران تعیین نمی‌کند. بازنشانی لایه‌ی تطبیق هدف، افزایش زیان طبقه‌بند بر پایه‌ی شیب و ترمیم پایانی بدون نقاب نیز بر وضعیت نهایی مدل اثر می‌گذارند. افزون بر این، لایه‌ی تطبیق مشترک نهایی همچنان به شیب‌های محاسبه‌شده از حافظه‌ی هدف وابسته است. در نتیجه، این مسیر هیچ تضمینی برای فراموشی دقیق یا فراموشی گواهی‌شده از نظر حریم خصوصی فراهم نمی‌کند.

\section{تحلیل نظری با دامنه‌ی محدود}
\label{sec:prop-theory}

این بخش فقط تغییر زیان مورد انتظار وظیفه‌های فعال را در زیرمرحله‌ی نقاب نرم بررسی می‌کند. بخش \ref{subsec:prop-theory-limits} توضیح می‌دهد که چرا نتیجه‌ی این تحلیل به افت عملکرد از آغاز تا پایان درخواست تعمیم نمی‌یابد و چگونه این محدودیت با یافته‌های تجربی فصل \ref{ch:results} سازگار است.

\subsection{نمادگذاری و افراز}
\label{subsec:prop-theory-notation}

در سراسر این تحلیل، بدنه‌ی منجمد $\theta_{\mathrm{base}}$ و لایه‌های تطبیق مخصوص وظیفه ثابت‌اند. بنابراین تنها پیراسنجه‌های متغیر و تنها منبع تداخل میان وظیفه‌ها در این زیرمرحله، وزن‌های لایه‌ی تطبیق مشترک $\phi\triangleq\phi_s\in\mathbb{R}^{d_s}$ هستند. $\phi^{(0)}$ مقدار این وزن‌ها در آغاز زیرمرحله و $\phi^{(N)}$ مقدار آن‌ها پس از $N$ گام است. هر درایه با رابطه‌ی زیر به‌روز می‌شود.
\begin{equation}
	\label{eq:coord-update}
	\phi^{(k+1)}_{i} = \phi^{(k)}_{i} - \eta\,m_{i}\,g^{(k)}_{i},
	\qquad
	g^{(k)}_{i} \triangleq \partial_{i}\,\mathcal{L}_{\mathrm{unlearn}}\big(\phi^{(k)}\big),
\end{equation}
ضریب $m_i=m_{\mathrm{soft}}(i)\in[0,1]$ در طول $N$ گام ثابت می‌ماند، زیرا نقاب فقط یک‌بار در مرحله‌ی یک محاسبه می‌شود. مجموعه‌ی اندیس‌ها به چهار ناحیه‌ی مجزا تقسیم می‌شود.
\begin{equation}
	\label{eq:partition}
	\begin{aligned}
		\mathcal{F}^{\circ} &\triangleq \Sforget\setminus\Sactive && (m_i = 1),\\
		\mathcal{S} &\triangleq \Scrit\setminus H && (m_i = 1-p),\\
		H &\subseteq \Scrit && (m_i = 0),\\
		\mathcal{N} &\triangleq \{1,\dots,d_s\}\setminus \Sforget && (m_i = 0).
	\end{aligned}
\end{equation}

\begin{definition}[زیان مورد انتظار وظیفه‌های نگه‌داری‌شده و افت زیرمرحله]
	\label{def:drop}
	برای هر وظیفه‌ی فعال $t\in\mathcal{T}_a$، زیان مورد انتظار آن بر حسب لایه‌ی تطبیق مشترک با $R_{t}(\phi)\triangleq\mathbb{E}_{(x,y)\sim D_t}[\ell(f(x;\phi),y)]$ تعریف می‌شود. افزایش این زیان در مسیر $\phi^{(0)}\!\to\!\phi^{(N)}$ برابر $\Delta_t\triangleq R_t(\phi^{(N)})-R_t(\phi^{(0)})$ است و افت عملکرد ناشی از همین زیرمرحله را نشان می‌دهد. $\Delta_t$ با بیشترین افت گزارش‌شده در فصل \ref{ch:results} تفاوت دارد، زیرا آن معیار کل فرایند را پوشش می‌دهد.
\end{definition}

\subsection{فرض‌ها}
\label{subsec:prop-theory-assumptions}

\begin{assumption}[همواری زیان مورد انتظار وظیفه‌های نگه‌داری‌شده]
	\label{asm:smooth}
	هر $R_t$ روی پوشش محدب مسیر، $\beta$-هموار است و در رابطه‌ی $\|\nabla R_t(\phi)-\nabla R_t(\phi')\|_2\leq\beta\|\phi-\phi'\|_2$ صدق می‌کند.
\end{assumption}

\begin{assumption}[کران‌داری شیب‌های فراموشی]
	\label{asm:bounded}
	$|g^{(k)}_i|\leq G$ به‌طور یکنواخت روی مسیر برقرار است. در نتیجه جابه‌جایی انباشته‌ی هر درایه از $\Gamma_i\triangleq|\sum_{k=0}^{N-1}g^{(k)}_i|\leq NG$ پیروی می‌کند.
\end{assumption}

\begin{assumption}[تمرکز حساسیت نگه‌داری روی ناحیه‌ی مشترک]
	\label{asm:concentration}
	$\epsilon\geq 0$ای وجود دارد که $|\partial_i R_t(\phi^{(0)})|\leq\epsilon$ برای هر $i\in\mathcal{F}^{\circ}$ و هر $t\in\mathcal{T}_a$.
\end{assumption}

فرض \ref{asm:concentration} شرطی اضافی درباره‌ی زیان مورد انتظار تک‌تک وظیفه‌های نگه‌داری‌شده است و فقط از قاعده‌ی انتخاب الگوریتم نتیجه نمی‌شود. الگوریتم درایه‌ها را با بزرگی شیب تجمیعی وظیفه‌های نگه‌داری‌شده رتبه‌بندی می‌کند. ممکن است شیب‌های وظیفه‌های مختلف یکدیگر را خنثی کنند و درایه‌ای که برای یک وظیفه مشتق بزرگی دارد، بیرون مجموعه‌ی انتخابی بماند. بنابراین لم بعدی تنها در صورت برقرار بودن این شرط معتبر است و اگر کوچک‌ترین مقدار معتبر $\epsilon$ بزرگ باشد، کران اطلاعات مفیدی نمی‌دهد.

\subsection{کران بالای افت زیرمرحله و یکنوایی آن نسبت به \texorpdfstring{$p$}{p}}
\label{subsec:prop-theory-prop}

\begin{lemma}[کران بالای افت زیرمرحله]
	\label{lem:subphase}
	فرض‌های \ref{asm:smooth} تا \ref{asm:concentration} برقرار هستند و $\eta>0$ و $N$ ثابت‌اند. برای هر وظیفه‌ی فعال $t$، {انرژی تداخل بحرانی\en{Critical Interference Energy}} با رابطه‌ی زیر تعریف می‌شود.
	\begin{equation}
		\label{eq:Ecrit}
		E^{\mathrm{crit}}_{t} \triangleq \eta\!\!\sum_{i\in \Scrit}\!\!\big|\partial_i R_t(\phi^{(0)})\big|\,\Gamma_i ,
	\end{equation}
	انرژی باقی‌مانده‌ی مخصوص هدف نیز با $C_t\triangleq\eta\sum_{i\in\mathcal{F}^{\circ}}\Gamma_i$ نشان داده می‌شود. در این صورت، به‌روزرسانی نقاب نرم با شدت محافظت $p$ از کران زیر پیروی می‌کند.
	\begin{equation}
		\label{eq:soft-bound}
		\Delta_t^{\mathrm{soft}} \leq (1-p)\,E^{\mathrm{crit}}_t + \epsilon\,C_t + \tfrac{\beta}{2}\,\eta^2 N^2 G^2\Big(|\mathcal{F}^{\circ}| + (1-p)^2|\mathcal{S}|\Big),
	\end{equation}
	در مقابل، به‌روزرسانی بی‌قید که در آن $m_i=1$ روی کل $\Sforget$ است، از کران زیر پیروی می‌کند.
	\begin{equation}
		\label{eq:unc-bound}
		\Delta_t^{\mathrm{unc}} \leq E^{\mathrm{crit}}_t + \epsilon\,C_t + \tfrac{\beta}{2}\,\eta^2 N^2 G^2\,|\Sforget| .
	\end{equation}
	سمت راست \eqref{eq:soft-bound} را \emph{کران جانشین} می‌نامیم و با $B_t(p)$ نشان می‌دهیم.
\end{lemma}

\begin{trivlist}
	\item[\hskip\labelsep{\em اثبات.}]
	اثبات را برای یک وظیفه‌ی فعال دلخواه $t$ انجام می‌دهیم و جابه‌جایی کل لایه‌ی تطبیق مشترک را $\delta\triangleq\phi^{(N)}-\phi^{(0)}$ می‌نامیم. با جمع‌بستن \eqref{eq:coord-update} روی $N$ گام و با توجه به ثابت بودن $m_i$، رابطه‌ی زیر به دست می‌آید.
	\begin{equation}
		\label{eq:disp}
		\delta_i = -\eta\,m_i\sum_{k=0}^{N-1}g^{(k)}_i
		\qquad\Longrightarrow\qquad
		|\delta_i|\leq\eta\,m_i\,\Gamma_i\leq\eta\,m_i\,NG,
	\end{equation}
	این رابطه از نابرابری مثلثی و فرض \ref{asm:bounded} به دست می‌آید.

	\textit{گام ۱ (بسط همواری).} طبق فرض \ref{asm:smooth}، کران بالای درجه‌دوم زیر برای $R_t$ در نقطه‌ی پایه‌ی $\phi^{(0)}$ برقرار است.
	\begin{equation}
		\label{eq:taylor}
		\Delta_t \leq \big\langle\nabla R_t(\phi^{(0)}),\,\delta\big\rangle + \tfrac{\beta}{2}\,\|\delta\|_2^2 .
	\end{equation}

	\textit{گام ۲ (جمله‌ی مرتبه‌ی اول).} افراز \eqref{eq:partition} و نابرابری مثلثی رابطه‌ی زیر را به دست می‌دهند.
	\begin{align}
		\big\langle\nabla R_t,\delta\big\rangle
		&\leq \sum_{i=1}^{d_s}\big|\partial_i R_t\big|\,|\delta_i| \notag\\
		&\leq \underbrace{\sum_{i\in\mathcal{F}^{\circ}}\!\big|\partial_i R_t\big|\,\eta\,\Gamma_i}_{m_i=1}
		+ \underbrace{\sum_{i\in\mathcal{S}}\!\big|\partial_i R_t\big|\,(1-p)\,\eta\,\Gamma_i}_{m_i=1-p}
		+ \underbrace{\sum_{i\in H\cup\mathcal{N}}\!\big|\partial_i R_t\big|\cdot 0}_{m_i=0}.
		\label{eq:three-sums}
	\end{align}
	جمله‌ی سوم همواره صفر است. طبق فرض \ref{asm:concentration}، جمله‌ی نخست حداکثر $\epsilon\,\eta\sum_{i\in\mathcal{F}^{\circ}}\Gamma_i=\epsilon C_t$ است. چون $\mathcal{S}\subseteq\Scrit$ و همه‌ی جمله‌های مجموع نامنفی‌اند، جمله‌ی دوم نیز حداکثر $(1-p)E^{\mathrm{crit}}_t$ است. بنابراین
	\begin{equation}
		\label{eq:first-order}
		\big\langle\nabla R_t,\delta\big\rangle \leq (1-p)\,E^{\mathrm{crit}}_t + \epsilon\,C_t .
	\end{equation}

	\textit{گام ۳ (جمله‌ی مرتبه‌ی دوم).} از \eqref{eq:disp} رابطه‌ی $|\delta_i|^2\leq\eta^2 m_i^2 N^2 G^2$ به دست می‌آید. همچنین $m_i^2=1$ روی $\mathcal{F}^{\circ}$، $m_i^2=(1-p)^2$ روی $\mathcal{S}$ و $m_i=0$ در دو ناحیه‌ی دیگر است. در نتیجه
	\begin{equation}
		\label{eq:second-order}
		\tfrac{\beta}{2}\|\delta\|_2^2 \leq \tfrac{\beta}{2}\,\eta^2 N^2 G^2\Big(|\mathcal{F}^{\circ}| + (1-p)^2|\mathcal{S}|\Big).
	\end{equation}
	جای‌گذاری \eqref{eq:first-order} و \eqref{eq:second-order} در \eqref{eq:taylor}، کران \eqref{eq:soft-bound} را می‌دهد.

	\textit{گام ۴ (خط پایه‌ی بی‌قید).} برای به‌روزرسانی بی‌قید، $m_i=1$ روی $\Sforget=\mathcal{F}^{\circ}\cup\Scrit$ و $m_i=0$ روی $\mathcal{N}$ است. با تکرار گام‌های ۲ و ۳ و استفاده از این ضرایب، جمله‌ی بحرانی بدون کاهش باقی می‌ماند و جمله‌ی انحنا برابر $\tfrac{\beta}{2}\eta^2N^2G^2|\Sforget|$ می‌شود. حاصل همان رابطه‌ی \eqref{eq:unc-bound} است.

\end{trivlist}

\begin{corollary}[کاهش کران جانشین با افزایش شدت محافظت]
	\label{cor:monotone}
	کران جانشین $B_t(p)$ از رابطه‌ی زیر به دست می‌آید.
	\begin{equation*}
		B_t(p)=(1-p)E^{\mathrm{crit}}_t+\epsilon C_t
		+\tfrac{\beta}{2}\eta^2N^2G^2\bigl(|\mathcal{F}^{\circ}|+(1-p)^2|\mathcal{S}|\bigr).
	\end{equation*}
	این تابع روی $p\in[0,1]$ مشتق‌پذیر است و با افزایش $p$ بزرگ‌تر نمی‌شود.
	\begin{equation}
		\label{eq:dBdp}
		\frac{\mathrm{d}B_t}{\mathrm{d}p} = -\,E^{\mathrm{crit}}_t \;-\; \beta\,\eta^2N^2G^2\,|\mathcal{S}|\,(1-p) \;\leq\; 0 .
	\end{equation}
	پس با بزرگ‌شدن $p$، کران کاهش می‌یابد و اندازه‌ی نرخ کاهش آن دست‌کم $E^{\mathrm{crit}}_t$ است. اگر $p=1$ باشد، یا به‌طور هم‌ارز $H$ کل $\Scrit$ را پوشش دهد، سهم ناحیه‌ی بحرانی از کران حذف می‌شود و فقط $\epsilon C_t$ و جمله‌ی انحنای مربوط به $\mathcal{F}^{\circ}$ باقی می‌ماند. همچنین $B_t(0)\leq B^{\mathrm{unc}}_t$ است و فقط در حالت $H=\varnothing$ تساوی برقرار می‌شود.
\end{corollary}

\begin{observation}[آنچه از این کران \emph{نتیجه نمی‌شود}]
	\label{obs:no-contraction}
	از $\Delta^{\mathrm{soft}}_t\leq B_t(p)$ و $\Delta^{\mathrm{unc}}_t\leq B^{\mathrm{unc}}_t$ و $B_t(p)\leq B^{\mathrm{unc}}_t$ نمی‌توان نتیجه گرفت که $\Delta^{\mathrm{soft}}_t\leq\Delta^{\mathrm{unc}}_t$ یا $\Delta^{\mathrm{soft}}_t\leq(1-p)\Delta^{\mathrm{unc}}_t$. مقایسه‌ی این دو کران بالا رابطه‌ای میان دو مقدار واقعی برقرار نمی‌کند. هر دو مقدار ممکن است بسیار پایین‌تر از کران خود باشند و ترتیبشان با ترتیب کران‌ها تفاوت داشته باشد. بنابراین نتیجه‌ی این بخش فقط به کران بالای نظری مربوط است، نه به افت واقعی.
\end{observation}

شکل \ref{fig:drop-decomposition} نشان می‌دهد که این کران جانشین با افزایش $p$ بزرگ‌تر نمی‌شود. بااین‌حال، کران \eqref{eq:soft-bound} را نمی‌توان به‌طور مستقیم با نتایج اجرا سنجید. سمت راست کران بر پایه‌ی زیان مورد انتظار است، در حالی که افت گزارش‌شده بر پایه‌ی دقت محاسبه می‌شود. افزون بر این، برای ثابت‌های $\epsilon$ و $\beta$ مقدار تجربی تعیین نشده است. حتی وجود همبستگی رتبه‌ای میان این دو کمیت نیز درستی کران را تأیید نمی‌کند و هیچ نتیجه‌ای در این پایان‌نامه بر چنین مقایسه‌ای تکیه ندارد.

	\centering

\subsection{چرا این کران افت کلی را پوشش نمی‌دهد}
\label{subsec:prop-theory-limits}

لم \ref{lem:subphase} فقط تغییرات زیرمرحله‌ی نقاب نرم را پوشش می‌دهد و به کل فرایند فراموشی تعمیم‌پذیر نیست. میان $\Delta_t$ در تعریف \ref{def:drop} و بیشترین افت گزارش‌شده در فصل \ref{ch:results} سه تفاوت وجود دارد.

\begin{enumerate}
	\item \textbf{بازنشانی پیش از زیرمرحله.} لایه‌ی تطبیق مخصوص وظیفه‌ی هدف و برش طبقه‌بند آن پیش از شروع زیرمرحله‌ی نقاب نرم بازنشانی می‌شوند. این تغییر در $\phi^{(0)}$ گنجانده شده است و تحلیل حاضر اثر آن را بررسی نمی‌کند، در حالی که ممکن است بخش عمده‌ی تغییر نهایی از همین مرحله ناشی شود.
	\item \textbf{صعود طبقه‌بند.} گام صعود مبتنی بر شیب روی برش طبقه‌بند هدف، بیرون از فضای $\phi_s$ عمل می‌کند و نقاب نرم آن را نمی‌بیند.
	\item \textbf{ترمیم بدون نقاب.} مرحله‌ی سه همه‌ی پیراسنجه‌های آموزش‌پذیر، از جمله درایه‌های $H$، را روی حافظه‌ی وظیفه‌های نگه‌داری‌شده به‌روز می‌کند. بنابراین حتی درایه‌هایی که در زیرمرحله‌ی پیشین ثابت بودند، ممکن است در پایان تغییر کرده باشند.
\end{enumerate}

این سه تفاوت نشان می‌دهند که کاهش کران جانشین با افزایش $p$، عملکرد نهایی گزارش‌شده را تضمین نمی‌کند. ازاین‌رو، لم \ref{lem:subphase} فقط سازوکار زیرمرحله را توضیح می‌دهد و پشتوانه‌ی نظری نتایج تجربی نیست. اثر واقعی نقاب نرم با حالت‌های مقایسه‌ای سنجیده می‌شود. نتایج بخش \ref{subsec:res-adapter-components} سود قابل‌اندازه‌گیری برای نقاب نرم در نقطه‌ی پایان نشان نمی‌دهند و سرکوب مشاهده‌شده را به بازنشانی مسیر هدف نسبت می‌دهند. پس کران نظری معتبر برای یک زیرمرحله‌ی محدود با نتیجه‌ی تجربی منفی برای کل فرایند تعارضی ندارد.

\section{ضوابط دسترسی به داده}
\label{sec:prop-data-access}

هر دو مسیر پیشنهادی یک محدودیت مشترک دارند. همه‌ی کمیت‌های مبتنی بر شیب که به داده‌های وظیفه‌ی هدف نیاز دارند، پیش از حذف نمونه‌های بازپخش آن محاسبه و ذخیره می‌شوند. پس از این مرحله، دسترسی تازه‌ای به داده‌های هدف انجام نمی‌شود. بررسی‌های خودکار این محدودیت را در ابتدا و انتهای بازه‌ی دسترسی کنترل می‌کنند و در صورت نقض آن، اجرا متوقف می‌شود.

فقط نمونه‌های وظیفه‌ی هدف از حافظه‌ی بازپخش مدل حذف می‌شوند. داده‌های خام روی دیسک، نقاط وارسی، نسخه‌های پشتیبان و نسخه‌های بیرونی در دامنه‌ی این فرایند نیستند. ظرفیت آزادشده‌ی حافظه‌ی بازپخش نیز میان وظیفه‌های باقی‌مانده توزیع نمی‌شود. در نتیجه، افزایش ظرفیت حافظه با اثر روش فراموشی اشتباه گرفته نخواهد شد.

\section{مقایسه‌ی دو مسیر و جمع‌بندی فصل}
\label{sec:prop-summary}

هر دو مسیر این فصل بر یک ایده استوارند. ابتدا ناحیه‌ی مشترک شناسایی می‌شود، سپس درایه‌های حساس به حفظ وظیفه‌های فعال از این ناحیه جدا و تغییرات آن‌ها محدود می‌شود. این دو مسیر از سه نظر تفاوت دارند. نخست، \lr{EPALL} ناحیه‌ی مشترک را از اشتراک دقیق دو نقاب دودویی ذخیره‌شده به دست می‌آورد، اما \lr{PALL-Adapter} از اشتراک دو مجموعه‌ی رتبه‌بندی‌شده بر پایه‌ی شیب استفاده می‌کند و به مقدار آستانه‌ها حساس است. دوم، \lr{EPALL} با افزودن جمله‌ی جریمه‌ی $\ell_2$ به تابع هدف، درایه‌های انتخاب‌شده را در سراسر حلقه‌ی ترمیم محافظت می‌کند. در مقابل، \lr{PALL-Adapter} اندازه‌ی گام را فقط در یک زیرمرحله کاهش می‌دهد. سوم، \lr{EPALL} روی کل ظرفیت شبکه عمل می‌کند، اما \lr{PALL-Adapter} تغییرات را به پیمانه‌های کوچکی محدود می‌کند که به یک بدنه‌ی منجمد افزوده شده‌اند.

همان‌طور که فصل بعد نشان می‌دهد، دو تفاوت نخست تعیین‌کننده‌اند. محافظت پایدار از یک ناحیه‌ی دقیق مؤثر است، اما مقیاس‌دهی موقت روی یک ناحیه‌ی تقریبی نتیجه‌ی نهایی را تغییر نمی‌دهد. بر پایه‌ی این نتیجه، \lr{EPALL} روش اصلی پایان‌نامه است و \lr{PALL-Adapter} فقط به‌عنوان یک گزینه‌ی اکتشافی و ارزیابی‌شده گزارش می‌شود.
