\chapter{آزمایش‌ها و نتایج}
\label{ch:results}

این فصل روش‌های فصل \ref{ch:proposed} را به‌طور تجربی ارزیابی می‌کند. نتایج بر پایه‌ی پرسش‌های پژوهش بخش \ref{sec:intro-questions} ارائه می‌شوند: ابتدا مقایسه‌ی اصلی و تحلیل جفتی آن (\lr{RQ1} و \lr{RQ2})، سپس آزمون موقعیت درخواست (\lr{RQ3}) و بررسی اجزای هر دو روش (\lr{RQ4} و \lr{RQ5})، و در پایان محاسبه‌ی هزینه، ارزیابی حریم خصوصی و بررسی حساسیت.

یک قاعده‌ی گزارش‌دهی در سراسر فصل رعایت می‌شود: هر جدول تعداد بذرهای خود را مشخص می‌کند، هیچ سلولی رتبه‌بندی نمی‌شود مگر آنکه معیار به‌تنهایی قابل تفسیر باشد، و نتایج با کمتر از پنج بذر صریحا \emph{توصیفی} برچسب می‌خورند. مسیر بازتولید خروجی‌های تجمیعی و برچسب‌های آزمایش در پیوست مستند شده است.

\section{طراحی آزمایش}
\label{sec:res-design}

\subsection{مجموعه‌داده‌های ارزیابی و تقسیم وظیفه‌ها}
\label{subsec:res-benchmarks}

سه مجموعه‌داده‌ی عمومی به‌کار می‌روند. این مجموعه‌داده‌ها به‌گونه‌ای به وظیفه‌ها تقسیم می‌شوند که تعداد وظیفه و دشواری بازنمایی به‌تدریج افزایش یابد.

\begin{itemize}
	\item \lr{CIFAR-10} \cite{krizhevsky2009learning} به‌صورت $5$ وظیفه‌ی دوکلاسه؛ سطح تصادفی $0.5$.
	\item \lr{CIFAR-100} \cite{krizhevsky2009learning} در طرح آزمایشی هم‌پوشانی‌سنگین به‌صورت $10$ وظیفه‌ی پنج‌کلاسه (سطح تصادفی $0.2$) و در طرح آزمایشی استاندارد به‌صورت وظیفه‌های ده‌کلاسه (سطح تصادفی $0.1$).
	\item \lr{TinyImageNet} \cite{le2015tinyimagenet} به‌صورت $20$ وظیفه‌ی ده‌کلاسه؛ سطح تصادفی $0.1$.
\end{itemize}

\subsection{سه طرح آزمایشی مجزا}
\label{subsec:res-protocols}

نکته‌ی روش‌شناختی مهم این است که آزمایش‌ها در سه طرح آزمایشی \emph{مجزا} اجرا شده‌اند و نتایج آن‌ها با یکدیگر مخلوط نمی‌شوند. تنها استثنا بررسی تنکی بخش \ref{sec:res-overlap-direct} است که افراز و دنباله‌ی طرح آزمایشی استاندارد را با بودجه‌ی $3$ دوره ترکیب می‌کند؛ همان‌جا تصریح شده که اعداد آن تنها درون خود جدول قابل مقایسه‌اند. سه طرح آزمایشی چنین‌اند:

\begin{description}
	\item[طرح آزمایشی استاندارد.] $20$ دوره به‌ازای هر وظیفه، بدنه‌ی آموزش‌دیده از صفر، هشت بذر $\{0,\ldots,7\}$. این طرح آزمایشی، مبنای مقایسه‌ی اصلی بخش \ref{subsec:res-main} است.
	\item[طرح آزمایشی هم‌پوشانی‌سنگین.] $3$ دوره به‌ازای هر وظیفه و افراز پنج‌کلاسه‌ی \lr{CIFAR-100}. این طرح آزمایشی عمدا هم‌پوشانی بیشتری بین زیرشبکه‌ها ایجاد می‌کند و برای کنترل‌های مؤلفه‌ای \lr{EPALL} و مقایسه‌ی سه نسخه‌ی \lr{PALL} به‌کار می‌رود.
	\item[طرح آزمایشی پیش‌آموزش‌دیده.] بدنه‌ی \lr{ImageNet} \cite{deng2009imagenet} منجمد. این طرح آزمایشی، فرض طبیعی روش‌های تنظیم کارآمد پیراسنجه‌ها را برقرار می‌کند و برای مقایسه‌ی \lr{LoRA} و \lr{PALL-Adapter} استفاده می‌شود.
\end{description}

در هر سه طرح آزمایشی، دنباله‌ی درخواست‌ها هر وظیفه را حداکثر یک‌بار آموزش می‌دهد، سه درخواست فراموشی دارد، و هیچ وظیفه‌ی حذف‌شده‌ای را دوباره معرفی نمی‌کند. درون هر بذر، همه‌ی روش‌ها \emph{دنباله‌ی درخواست یکسانی} می‌بینند؛ همین جفت‌شدگی است که تحلیل جفتی بخش \ref{subsec:res-paired} را ممکن می‌کند.

\subsection{معماری‌ها و ابرپیراسنجه‌ها}
\label{subsec:res-hyper}

به‌جز آزمون \lr{ViT}، اجراهای \lr{CIFAR-10}، \lr{TinyImageNet} و پیش‌آموزش‌دیده از نسخه‌های \lr{ResNet-18} و اجراهای اصلی \lr{CIFAR-100} از نسخه‌های \lr{ResNet-34} استفاده می‌کنند \cite{he2016deep}. بهینه‌ساز \lr{SGD} \cite{robbins1951sgd} با نرخ یادگیری $0.01$، تکانه‌ی $0.9$، تنظیم وزن $5\times10^{-4}$ و اندازه‌ی دسته‌ی $32$ است. بودجه‌ی بازپخش $500$ نمونه و از پیش میان وظایف افراز می‌شود، بدون بازتوزیع پس از حذف. تنها استثنا، \lr{LoRA} روی \lr{Split-CIFAR-100} است که نرخ یادگیری $10^{-3}$ می‌گیرد، چون با $10^{-2}$ واگرا می‌شود.

پیکربندی روش‌های پیشنهادی چنین است. \lr{EPALL}: نسبت محافظت $\rho=0.2$ و ضریب لنگر $\lambda=1$. \lr{PALL-Adapter}: گلوگاه وظیفه و گلوگاه مشترک $16$، $\alpha_f=0.3$، $\alpha_p=0.2$، $p=|\Scrit|/|\Sforget|$، ده گام فراموشی و پنجاه گام ترمیم. رتبه‌ی \lr{LoRA} برابر $8$. تنکی زیرشبکه $0.8$ روی \lr{CIFAR-10} و \lr{TinyImageNet} و $0.9$ روی \lr{CIFAR-100} است. پیکربندی کامل آزمایش‌ها در جدول \ref{tab:res-hyper} در پیوست آمده است.

\subsection{بذرها و سخت‌افزار}
\label{subsec:res-seeds}

مقایسه‌ی اصلی، حالت‌های آزمایشی مستقیم \lr{EPALL} و تحلیل لنگر از بذرهای $\{0,\ldots,7\}$ استفاده می‌کنند. جدول‌های پیش‌آموزش‌دیده، هم‌پوشانی زیاد و اجزای لایه‌ی تطبیق بر پایه‌ی بذرهای $\{0,1,2\}$ هستند و تحلیل معیار اهمیت، \lr{ViT} و ماندگاری از بذرهای $\{0,1\}$ استفاده می‌کنند. آزمون موقعیت درخواست و بررسی تنکی زیرشبکه هر دو پنج بذر $\{0,\ldots,4\}$ دارند و بررسی شدت نقاب پیوسته با سه بذر انجام شده است. مرجع بازآموزی با دو بذر ارزیابی شده است. ارزیابی اصلاح‌شده‌ی \lr{MIA} برای \lr{EPALL} سه بذر و برای \lr{PALL} اصلی دو بذر دارد. حافظه نیز روی یک دنباله‌ی هم‌تراز محاسبه شده است. بررسی عرض گلوگاه با دو بذر فقط برای نشان‌دادن نبود روند یکنواخت به‌کار می‌رود. پیکربندی سخت‌افزاری و نرم‌افزاری اجراها، نسخه‌ی مخزن و درهم‌سازی دنباله‌های درخواست در پیوست آمده‌اند.

\section{معیارهای ارزیابی}
\label{sec:res-metrics}

\paragraph{دقت نهایی نگه‌داری $\Afinal$ (بیشتر بهتر).} میانگین دقت روی وظایف فعال در پایان دنباله.

\paragraph{فراموشی میانگین $\Favg$ (کمتر بهتر).} برای درخواست $r$:
\begin{equation}
	\label{eq:favg}
	\Favg(r)=\frac{1}{|\mathcal{T}_{\mathrm{trained}}|}\sum_{t}\Big[\max_{k<r}A_t(k)-A_t(r)\Big].
\end{equation}
این معیار روی همه‌ی وظایفی محاسبه می‌شود که تا لحظه‌ی $r$ آموزش دیده‌اند، از جمله وظایفی که بعدا حذف شده‌اند. به همین دلیل $\Favg$ وضعیت \emph{کل مسیر} یادگیری پیوسته را می‌سنجد و نه اثر یک درخواست مشخص را؛ برای سنجش اثر یک درخواست از دو معیار بعدی استفاده می‌شود.

\paragraph{افت نگه‌داری $F_u$ و بیشترین افت (کمتر بهتر).} $F_u$ اختلاف میانگین دقت وظایف نگه‌داری‌شده پیش و پس از یک درخواست است، و
\begin{equation}
	\label{eq:worstdrop}
	\mathrm{WorstDrop} = \max_{t \in \mathcal{T}_{a}} \big( A_{t}^{\mathrm{before}} - A_{t}^{\mathrm{after}} \big).
\end{equation}
این معیار \emph{علامت‌دار} گزارش می‌شود و در صفر بریده نمی‌شود؛ بنابراین مقدار منفی به این معنا است که ترمیم، بدترین وظیفه را بهبود داده است و این اطلاعات با بریدن از دست می‌رفت.

\paragraph{دقت وظیفه‌ی فراموش‌شده $A_u$.} بی‌درنگ پس از درخواست اندازه‌گیری می‌شود و همیشه از طریق $|A_u-c|$ خوانده می‌شود که $c$ سطح تصادفی همان وظیفه است. تأکید می‌شود که این معیار «کمتر بهتر» نیست.

\paragraph{زمان درخواست $T_f$ و نسبت دامنه‌ی ترمیم $\Rupd$.} $T_f$ زمان پردازش درخواست است و با \lr{\texttt{time.perf\_counter}} اندازه‌گیری می‌شود. چون فراخوانی‌های \lr{CUDA} ناهم‌گام‌اند و پیش از خواندن ساعت \lr{\texttt{torch.cuda.synchronize}} اجرا نمی‌شود، این مقدار ممکن است همه‌ی کارهای قرارگرفته در صف را در بر نگیرد. بنابراین، $T_f$ فقط برای مقایسه‌ی تقریبی روش‌ها روی همین محیط اجرا مناسب است و زمان دقیق محاسبه را نشان نمی‌دهد. $\Rupd$ فقط نسبت اسمی پیراسنجه‌های آموزش‌پذیر در نقاب ترمیم را نشان می‌دهد و کل محاسبات، حافظه، زمان اجرا یا تعداد واقعی وزن‌های تغییریافته را اندازه نمی‌گیرد. به همین دلیل، بخش \ref{sec:res-storage} حافظه‌ی موردنیاز را جداگانه محاسبه می‌کند. جدول‌ها بر پایه‌ی آخرین رویداد فراموشی هر اجرا تهیه شده‌اند و میانگین و انحراف معیار نمونه‌ای روی بذرها گزارش می‌شوند.

\subsection{تفسیر نتایج اصلی}
\label{subsec:res-main-read}

\paragraph{\lr{CLPU} قوی‌ترین نتیجه را دارد و ادعای برتری بر آن مطرح نمی‌شود.} پیاده‌سازی ارزیابی‌شده برای هر وظیفه‌ی فعال یک شبکه‌ی کامل نگه می‌دارد و فراموشی را با کنارگذاشتن شبکه‌ی همان وظیفه انجام می‌دهد. بنابراین، زمان درخواست کمتر از یک میلی‌ثانیه فقط هزینه‌ی حذف آن شبکه از ساختار داده را نشان می‌دهد و هزینه‌ی نگه‌داری شبکه‌های جداگانه را در بر نمی‌گیرد. \lr{CLPU} در صورت پذیرفتنی‌بودن این جداسازی گزینه‌ی مناسبی است، اما از یک مدل مشترک استفاده نمی‌کند. بخش \ref{sec:res-storage} حافظه‌ی موردنیاز آن را گزارش می‌کند.

\paragraph{میان روش‌های مدل‌مشترک، \lr{EPALL} به مرجع جداسازی نزدیک می‌شود.} روی \lr{CIFAR-10} فاصله‌ی $\Afinal$ آن از \lr{CLPU} کمتر از یک واحد درصد است ($0.9433$ در برابر $0.9500$) و روی \lr{CIFAR-100} دو مقدار از نظر عددی نزدیک‌اند ($0.7371$ در برابر $0.7355$). این نزدیکی نه برتری است و نه هم‌ارزی اثبات‌شده: اختلاف نسبت به پراکندگی بذرها کوچک است و هیچ آزمون هم‌ارزی برای آن اجرا نشده است. هم‌زمان $A_u$ را به سطح تصادفی می‌رساند ($0.4863$ و $0.0959$) و بیشترین افت را در $0.0027$ و $0.0056$ نگه می‌دارد. بنابراین بدون دادن یک شبکه‌ی کامل به هر وظیفه، عملا به همان نقطه می‌رسد.

\paragraph{ستون $A_u$ روش‌های حفظ‌محور را رد می‌کند.} \lr{EWC} و \lr{LwF} روی \lr{CIFAR-10} بیشترین افت صفر ثبت می‌کنند — که در نگاه اول عالی است — اما $A_u$ آن‌ها $0.8122$ و $0.9469$ است؛ یعنی وظیفه‌ی هدف عملا حذف نشده و بخش عمده‌ی رفتار آن حفظ شده است. به همین ترتیب، \lr{SSD} روی \lr{CIFAR-10} با $A_u=0.8453$ می‌ماند و هر دو روش \lr{SSD} و \lr{SalUn} روی \lr{CIFAR-100} با $0.2619$ و $0.3967$ به‌روشنی بالای هدف $0.1$ باقی می‌مانند. این دقیقا نمونه‌ای است که چرا معیارها باید هم‌زمان خوانده شوند: هیچ‌یک از این دو ستون به‌تنهایی داوری درستی نمی‌دهد.

\paragraph{دو سطر \lr{PEFT} تشخیصی‌اند و رقیب هم‌ظرفیت نیستند.} در این طرح آزمایشی از صفر، هم \lr{LoRA} و هم \lr{PALL-Adapter} روی یک \lr{ResNet} منجمد با مقدارگذاری تصادفی نشسته‌اند. دقت \lr{CIFAR-100} آن‌ها به‌طور مشابه پایین است ($0.2695$ و $0.2583$) به‌رغم آنکه سازوکار فراموشی‌شان متفاوت است، و هر دو با پیش‌آموزش‌دیده‌شدن بدنه بازیابی می‌شوند (جدول \ref{tab:res-pretrained}). این الگو به یک گلوگاه \emph{بازنمایی} اشاره می‌کند و نه به گام فراموشی.

روی \lr{CIFAR-100} اختلاف $\Afinal$ مثبت و برابر $+0.0147$ با بازه‌ی $[0.0062,0.0229]$ است، با بهبود میانگین $0.0110$ در $\Favg$ و $0.0166$ در بیشترین افت؛ هیچ‌یک از این سه بازه صفر را در بر نمی‌گیرد. روی \lr{CIFAR-10} افزایش دقت $+0.0087$ $[0.0034,0.0151]$ و بهبود بیشترین افت $0.0092$ $[0.0029,0.0157]$ است، در حالی که $\Favg$ نتیجه‌ی مشخصی ندارد ($-0.0017$ با بازه‌ای که صفر را در بر می‌گیرد). هزینه‌ی این نگه‌داری، زمان درخواست است: \lr{EPALL} حدود $2.10$ ثانیه روی \lr{CIFAR-10} و $12.06$ ثانیه روی \lr{CIFAR-100} از \lr{PALL} اصلی کندتر است.

جدول \ref{tab:res-significance} آزمون‌های دقیق یک‌طرفه را گزارش می‌کند. چون شش آزمون هم‌زمان انجام می‌شود، مقدار $p$ خام به‌تنهایی کافی نیست و اصلاح \lr{Holm} برای چندگانگی اعمال شده است. روی \lr{CIFAR-100} هر سه معیار به $p=0.007812$ می‌رسند و هفت جفت از هشت جفت به سود \lr{EPALL} است، با اندازه‌های اثر جفتی $d_z$ برابر $1.11$، $1.51$ و $1.35$؛ هر سه پس از اصلاح \lr{Holm} نیز در سطح $0.05$ باقی می‌مانند. روی \lr{CIFAR-10} تنها $\Afinal$ باقی می‌ماند ($p=0.007812$ و پس از \lr{Holm} برابر $0.0469$).

\input{tables/res_significance}

تفسیر این جدول باید محتاطانه بماند و دو تصحیح مهم دارد. نخست، \textbf{بیشترین افت روی \lr{CIFAR-10} پس از اصلاح \lr{Holm} معنادار نیست}: مقدار خام $0.03125$ پس از اصلاح به $0.0625$ می‌رسد و از سطح $0.05$ می‌گذرد. آزمون علامت همان معیار نیز به $p=0.109$ می‌رسد، پس شاهد آن از دو جهت ضعیف است و در این پایان‌نامه به‌عنوان یافته‌ی معنادار گزارش نمی‌شود. دوم، \textbf{کف $p$ قابل‌دستیابی برای همه‌ی سطرها یکسان نیست}: این کف به تعداد جفت‌های بدون تساوی بستگی دارد و برای سطرهای \lr{CIFAR-100} (بدون تساوی) برابر $0.003906$ است، نه $0.007812$. بنابراین مقادیر مشاهده‌شده‌ی \lr{CIFAR-100} \emph{روی کف نایستاده‌اند} بلکه دو برابر آن هستند، و این یعنی توان آزمون کاملا اشباع نشده است. تنها معیاری که در هیچ مجموعه‌داده‌ای بهبود پیوسته نشان نمی‌دهد $\Favg$ روی \lr{CIFAR-10} است، که با علامت منفی اختلاف جفتی در جدول \ref{tab:res-paired} سازگار است.

در مجموع، از شش آزمون، \textbf{چهار آزمون} پس از اصلاح هولم برای آزمون‌های متعدد همچنان معنادارند: هر سه معیار روی \lr{CIFAR-100} و دقت نهایی روی \lr{CIFAR-10}.

\paragraph{یادداشت درباره‌ی تعداد بذر.} نتایج این بخش بر هشت بذر ($0$ تا $7$) استوارند که وضعیت کامل‌شده‌ی مخزن نتایج است. نسخه‌ی مقاله‌ی کنفرانسی همین کار بر پنج بذر ($0$ تا $4$) بسته شده بود و اعداد آن با اعداد این‌جا تفاوت جزئی دارد؛ جهت همه‌ی نتیجه‌گیری‌ها یکسان مانده است، اما مقادیر عددی و $p$ها متفاوت‌اند. هرجا در این پایان‌نامه عددی گزارش می‌شود، منبع آن وضعیت کنونی مخزن نتایج است و نه مقاله.

\section{آزمون موقعیت درخواست}
\label{sec:res-overlap-response}

فرض اصلی \lr{EPALL} این است که با افزایش هم‌پوشانی، محافظت از وزن‌های مشترک اهمیت بیشتری پیدا می‌کند. برای بررسی این فرض، آزمونی با پنج سطح از پیش تعیین‌شده طراحی شد: پنج سطح $\times$ شش روش $\times$ دو مجموعه‌داده $\times$ پنج بذر $=300$ اجرای هم‌تراز.

\textbf{این آزمون، هم‌پوشانی را مستقیما کنترل نمی‌کند.} متغیری که میان سطوح تغییر می‌کند، \emph{موقعیت درخواست فراموشی در دنباله} است؛ یعنی تعداد وظیفه‌هایی که پس از وظیفه‌ی هدف آموزش می‌بینند. این کمیت با $K$ نشان داده می‌شود: برای \lr{CIFAR-10} برابر $K\in\{0,1,2,3,4\}$ و برای \lr{CIFAR-100} برابر $K\in\{0,2,4,7,9\}$ است. چون تقسیم کلاس‌ها و شناسه‌های وظیفه ثابت‌اند، کنترل مستقیم هم‌پوشانی نقاب‌ها به تغییر تقسیم داده نیاز دارد که در این پژوهش انجام نشده است.

در نتیجه، $K$ فقط نشانه‌ای تقریبی از هم‌پوشانی موردانتظار است. با تغییر آن، هویت وظیفه‌ی هدف، زمان سپری‌شده از آموزش آن، ترتیب درخواست‌ها و مقدار آموزش پس از آن نیز هم‌زمان تغییر می‌کنند. این آزمون در سراسر پایان‌نامه \textbf{آزمون موقعیت درخواست} نامیده می‌شود و نتیجه‌ی آن فقط روندی توصیفی و سازگار با فرضیه است، نه شاهدی برای اثر مستقل هم‌پوشانی. جدول \ref{tab:res-overlap-levels} بیشترین افت را در همه‌ی حالت‌ها گزارش می‌کند.

\input{tables/res_overlap_levels}

الگوی \lr{CIFAR-100} همان چیزی است که سازوکار پیش‌بینی می‌کرد: بیشترین افت \lr{PALL} اصلی با افزایش شاخص موقعیت درخواست از $0.0124$ به $0.0456$ رشد می‌کند، در حالی که \lr{EPALL} در بازه‌ی $0.0060$ تا $0.0164$ نسبتا ثابت می‌ماند. روی \lr{CIFAR-10} همین ترتیب حفظ می‌شود اما در بازه‌ای کوچک‌تر. \lr{SalUn} رفتاری مشابه \lr{PALL} اصلی نشان می‌دهد و \lr{CLPU} همان‌گونه که انتظار می‌رفت به‌طور ساختاری صفر می‌ماند، چون شبکه‌های مستقل امکان تداخل ندارند. شکل \ref{fig:res-overlap-response} همین روند را نمایش می‌دهد.

\section{بررسی تنکی و هم‌پوشانی اندازه‌گیری‌شده}
\label{sec:res-overlap-direct}

آزمون بخش \ref{sec:res-overlap-response} محدودیتی دارد که در همان بخش توضیح داده شد. سطوح آزمایش موقعیت درخواست را تغییر می‌دهند و هم‌پوشانی فقط به‌عنوان پیامد این تغییر جابه‌جا می‌شود. هویت وظیفه‌ی هدف، زمان سپری‌شده از آموزش آن و مقدار آموزش پس از آن نیز هم‌زمان تغییر می‌کنند. بنابراین، نتیجه‌ی آن آزمون با فرضیه سازگار است، اما آن را مستقیما نمی‌آزماید. در این بخش، موقعیت درخواست ثابت می‌ماند و تنکی زیرشبکه تغییر می‌کند. چون تغییر تنکی ظرفیت را نیز تغییر می‌دهد، این آزمایش هم اثر مستقل هم‌پوشانی را اندازه نمی‌گیرد.

\paragraph{طرح.} دنباله‌ی درخواست، وظیفه‌ی هدف و موقعیت آن دقیقا ثابت نگه داشته می‌شوند و تنها \emph{تنکی زیرشبکه} تغییر می‌کند. تنکی تعیین می‌کند هر نقاب وظیفه چند درایه بگیرد و در نتیجه با $|\Sshare|=|\Mf\cap\Mr|$ ارتباط مستقیم دارد، اما هم‌زمان ظرفیت هر وظیفه را نیز تغییر می‌دهد. پنج سطح تنکی از $0.9$ تا $0.5$، دو روش و پنج بذر، در مجموع $100$ اجرای هم‌تراز می‌سازند. این بررسی افراز و دنباله‌ی طرح آزمایشی استاندارد را به‌کار می‌برد، اما با بودجه‌ی $3$ دوره به‌ازای هر وظیفه اجرا شده است و نه $20$ دوره. بنابراین سطح مطلق دقت آن با جدول‌های بخش \ref{subsec:res-main} قابل مقایسه نیست و تنها مقایسه‌ی \emph{درون‌جدولی} دو حالت آزمایشی در هر سطح تنکی معتبر است. دو حالت آزمایشی هر خانه و بذر یکسان می‌بینند؛ هم‌پوشانی اندازه‌گیری‌شده‌ی آن‌ها کاملا برابر نیست، ازاین‌رو میانگین جفت گزارش می‌شود. بیشترین اختلاف نسبی دو حالت آزمایشی کمتر از $0.05\%$ است.

\paragraph{آزمون تنکی.} برخلاف آزمون موقعیت درخواست که متغیر مستقل آن فقط سه یا چهار مقدار متفاوت داشت، هم‌پوشانی اندازه‌گیری‌شده در اینجا روی \lr{CIFAR-10} به اندازه‌ی $19.7$ برابر و روی \lr{CIFAR-100} به اندازه‌ی $10.7$ برابر تغییر می‌کند و با سطوح تنکی رابطه‌ای یکنواخت دارد. بنابراین، تنکی دامنه‌ی مناسبی از هم‌پوشانی ایجاد کرده است؛ بااین‌حال، اثر هم‌پوشانی را از اثر ظرفیت جدا نمی‌کند.

\input{tables/res_overlap_sparsity}

جدول \ref{tab:res-overlap-sparsity} نتیجه را نشان می‌دهد. روی \lr{CIFAR-10} سود حفاظت از $0.0018$ در کم‌ترین هم‌پوشانی به $0.0349$ در بیش‌ترین می‌رسد. روی \lr{CIFAR-100} از $0.0146$ آغاز می‌شود، در سطح تنکی $0.6$ به بیشینه‌ی $0.1002$ می‌رسد و در بیشترین هم‌پوشانی $0.0794$ است؛ بنابراین روند مثبت است اما یکنواخت نیست. همبستگی رتبه‌ای اسپیرمن میان هم‌پوشانی اندازه‌گیری‌شده و سود حفاظت روی \lr{CIFAR-10} برابر $+1.00$ و روی \lr{CIFAR-100} برابر $+0.90$ است. چون تنها پنج سطح در کار است، به‌جای تقریب مجانبی، هر $120$ جایگشت ممکن شمرده شده و $p$ دقیق به‌دست آمده است: به‌ترتیب $0.0083$ و $0.0417$. برای سود دقت نهایی، همبستگی روی هر دو مجموعه‌داده $+1.00$ و $p$ دقیق $0.0083$ است.

\paragraph{دو قید.} نخست، تنکی علاوه بر هم‌پوشانی، \emph{ظرفیت} را هم تغییر می‌دهد و این عامل مداخله‌گر با طراحی حذف‌شدنی نیست: دقت \lr{PALL} در همین بررسی روی \lr{CIFAR-100} از $0.4919$ به $0.3292$ می‌افتد. همسان‌بودن تنکی اسمی در هر جفت، مقایسه‌ی دو روش را تفسیرپذیرتر می‌کند، اما اثر علی مستقل هم‌پوشانی را شناسایی نمی‌کند. دوم، آزمون علامت میان بذرها درون هر خانه با پنج بذر توان کمی دارد و تنها سه خانه از ده خانه به $p<0.05$ می‌رسند؛ میانگین‌ها اختلاف بزرگی نشان می‌دهند اما پراکندگی بذرها بالاست. بنابراین شاهد اصلی این بخش در روند میان سطوح است و نه در تک‌تک خانه‌ها.

\paragraph{جمع‌بندی.} با ثابت‌نگه‌داشتن هویت و موقعیت وظیفه‌ی هدف، بررسی تنکی یک رابطه‌ی رتبه‌ای مثبت میان هم‌پوشانی اندازه‌گیری‌شده و سود حفاظت نشان می‌دهد. این نتیجه از ادعای انگیزشی \lr{EPALL} پشتیبانی می‌کند، اما به‌دلیل تغییر هم‌زمان ظرفیت و یکنواخت‌نبودن روند \lr{CIFAR-100} اثبات علی خالص اثر هم‌پوشانی نیست.

تفسیر این دو جدول چهار نتیجه می‌دهد. دو نتیجه به سود سازوکار پیشنهادی هستند و دو نتیجه چنین نیستند؛ هر چهار به‌صراحت گزارش می‌شوند.

\paragraph{یک. کیفیت انتخاب مهم است، نه فقط اندازه‌ی بودجه.} این روشن‌ترین و پایدارترین یافته‌ی کنترل‌ها است و روی \emph{هر دو} مجموعه‌داده برقرار می‌ماند. حالت آزمایشی بودجه‌ی تصادفی همان تعداد درایه را محافظت می‌کند، اما بیشترین افت را $0.0076$ روی \lr{CIFAR-10} و $0.0148$ روی \lr{CIFAR-100} بدتر می‌کند و هر دو بازه صفر را در بر نمی‌گیرند؛ روی \lr{CIFAR-10} اختلاف دقت نیز ($+0.0027$) از صفر جدا است. این مقایسه دقیقا همان چیزی را جدا می‌کند که باید: با ثابت‌نگه‌داشتن اندازه‌ی بودجه، تفاوت باقی‌مانده تنها می‌تواند از کیفیت رتبه‌بندی بیاید.

\paragraph{دو. رتبه‌بندی نسبت به محافظت بدون رتبه، نگه‌داری بهتری می‌دهد اما در همه‌ی معیارها برتر نیست.} روی \lr{CIFAR-100}، حالت کامل دقت جفتی را نسبت به حالت فقط هم‌پوشانی به اندازه‌ی $0.0120$ با بازه‌ی $[0.0065,0.0191]$ بهبود می‌دهد. بااین‌حال، حالت فقط هم‌پوشانی روی هر دو مجموعه‌داده بیشترین افت کمتری دارد: $0.0022$ روی \lr{CIFAR-10} و $0.0075$ روی \lr{CIFAR-100}، با بازه‌هایی که صفر را در بر نمی‌گیرند. بنابراین، میان حفظ بهتر دقت نهایی و کاهش افت آسیب‌پذیرترین وظیفه موازنه وجود دارد. اگر مهم‌ترین معیار سامانه کمینه‌کردن افت بدترین وظیفه باشد، محافظت بدون رتبه‌بندی انتخاب بهتری است.

\paragraph{سه. سهم مستقل جریمه‌ی $\ell_2$ پشتیبانی نمی‌شود.} این حالت آزمایشی تنها جمله‌ی جریمه را حذف می‌کند؛ شناسایی ناحیه‌ی بحرانی و ساخت پشتیبان‌های آن دست‌نخورده باقی می‌ماند، پس نتیجه‌ی آن درباره‌ی \emph{شدت مهار} است و نه درباره‌ی کل محافظت. این حالت آزمایشی روی هیچ‌یک از دو مجموعه‌داده به‌روشنی از حالت کامل جدا نمی‌شود: اختلاف دقت جفتی روی \lr{CIFAR-100} برابر $0.0052$ با بازه‌ی $[-0.0008,0.0129]$ است که صفر را در بر می‌گیرد، و روی \lr{CIFAR-10} برآورد نقطه‌ای $-0.0048$ است، یعنی حتی کمی به سود \emph{حذف} لنگر. اختلاف بیشترین افت در هر دو مورد عملا صفر است. بنابراین در پیکربندی آزمون‌شده، آنچه اثر دارد \emph{انتخاب} ناحیه‌ی محافظت‌شده است و نه شدت مهار آن با جمله‌ی $\ell_2$. این یافته با نسخه‌ی مبتنی بر پنج بذر همین کنترل‌ها تفاوت دارد، جایی که بازه‌ی \lr{CIFAR-100} صفر را در بر نمی‌گرفت؛ افزودن بذر آن جدایی را از میان برد.

\paragraph{چهار. فیلتر صریح اشتراک در \emph{این پیاده‌سازی} اضافی است.} حالت آزمایشی رتبه‌بندی بدون اشتراک، حالت کامل را \emph{دقیقا} بازتولید می‌کند؛ همه‌ی اختلاف‌های جفتی صفرند. اما تفسیر این یکسانی باید دقیق باشد و \textbf{ضرورت عمومی هم‌پوشانی را رد نمی‌کند}. دلیلش ساختاری است: در این حالت آزمایشی رتبه‌بندی روی کل زیرشبکه‌ی هدف $\Mf$ انجام می‌شود، ولی نقاب ترمیم \emph{مستقل از حالت کنترل} همان $\Mf\cap\Mr$ می‌ماند و شیب بیرون آن صفر می‌شود. پس هر درایه‌ای که رتبه‌بندی بیرون $\Mr$ انتخاب کند، در عمل هرگز حرکت نمی‌کند و لنگر روی آن بی‌اثر است. به بیان دیگر، خود پیاده‌سازی فیلتر اشتراک را در مرحله‌ی بعد اعمال می‌کند؛ بنابراین این آزمایش نشان می‌دهد اعمال \emph{دوباره}ی آن در مرحله‌ی پیش اضافی است، نه اینکه محدودکردن به ناحیه‌ی مشترک بی‌فایده باشد.

\paragraph{جمع‌بندی کنترل‌ها.} آنچه این کنترل‌ها به‌طور تجربی پشتیبانی می‌کنند، باریک‌تر از توصیف کامل سازوکار است: \emph{رتبه‌بندی درایه‌های نامزد بر پایه‌ی حساسیت نگه‌داری، با یک بودجه‌ی ثابت}. دو جزء دیگر سهم مستقل قابل‌اندازه‌گیری نشان نمی‌دهند، اما این به‌معنای بی‌فایده‌بودن آن‌ها نیست: جریمه‌ی $\ell_2$ تنها شدت مهار را تنظیم می‌کند در حالی که پشتیبان‌ها مستقل از آن ساخته می‌شوند، و فیلتر اشتراک در مرحله‌ی بعد پیاده‌سازی به‌هرحال اعمال می‌شود. این کنترل‌ها برای اثبات \emph{ضرورت} همه‌ی اجزای طراحی کافی نیستند. این محدودیت در بخش \ref{sec:concl-limitations} به‌عنوان محدودیت ادعای سازوکاری ثبت می‌شود و در بخش \ref{sec:concl-future} به یک پیشنهاد ساده‌سازی تبدیل می‌شود.

\paragraph{نقاب نرم در نقطه‌ی پایان بی‌اثر است.} حالت کامل و حالت آزمایشی یکنواخت حداکثر $0.0006$ در $\Afinal$ یا بیشترین افت با هم اختلاف دارند، و حالت کامل و نقاب بدون صعود در هر دو معیار \emph{یکسان}اند. چون حالت‌های آزمایشی از نظر مولد اعداد تصادفی هم‌تراز طراحی شده‌اند، این یکسانی نشانه‌ی نوسان تصادفی نیست؛ نشانه‌ی یک مؤلفه‌ی واقعا بی‌اثر است.

\paragraph{صعود طبقه‌بند تنها مؤلفه‌ی با اثر قابل‌اندازه‌گیری است.} حذف آن، نگه‌داری را دست‌نخورده می‌گذارد اما وظیفه‌ی فراموش‌شده را به‌اندازه‌ی $0.0040$ روی \lr{CIFAR-100} و $0.0017$ روی \lr{CIFAR-10} از سطح تصادفی دور می‌کند. یعنی این مؤلفه در سرکوب سهم دارد، هرچند کوچک، و به دقت نگه‌داری آسیب نمی‌زند.

\paragraph{روی \lr{CIFAR-100} فقط بازنشانی در واقع \emph{بهتر} است.} نگه‌داری $0.8324$ در برابر $0.8244$، بیشترین افت $0$ در برابر $0.0147$، و فاصله‌ی میانگین کوچک‌تری تا سطح تصادفی. بازه‌های بازنمونه‌گیری جفتی روی سه دنباله برای اختلاف دقت و اختلاف فاصله تا سطح تصادفی صفر را در بر می‌گیرند، در حالی که اختلاف بیشترین افت به سود فقط بازنشانی است.

نتایج نشان می‌دهند که سرکوب مشاهده‌شده در \lr{PALL-Adapter} عمدتا از \emph{بازنشانی لایه‌ی مخصوص وظیفه‌ی هدف} ناشی می‌شود. به‌روزرسانی آگاه از هم‌پوشانی لایه‌ی مشترک، با وجود تحلیل نظری محدود در بخش \ref{sec:prop-theory}، نتیجه‌ی نهایی را به‌طور قابل‌اندازه‌گیری بهبود نمی‌دهد. به همین دلیل، این روش فقط به‌عنوان یک گزینه‌ی اکتشافی و ارزیابی‌شده گزارش می‌شود و مشارکت اصلی پایان‌نامه نیست.

\subsection{نقاب نرم پیوسته: بررسی شدت}
\label{subsec:res-conflict-gamma}

نتیجه‌ی منفی بخش پیش، نقاب نرم \emph{گسسته} را ارزیابی می‌کند: افراز دودویی کامل/نرم/منجمد با یک ضریب ثابت. یک تبیین محتمل برای آن نتیجه این است که مشکل نه در ایده‌ی محافظت، بلکه در \emph{درشت‌دانگی} افراز باشد. برای آزمودن این تبیین، نسخه‌ی پیوسته‌ای پیاده‌سازی و در چند سطح بررسی شد که در آن هر درایه‌ی زیرمجموعه‌ی فراموشی ضریب خاص خود را می‌گیرد:
\begin{equation}
\label{eq:continuous-mask}
	m_i=\mathrm{clamp}\big(1-\gamma\,\hat c_i,\;0,\;1\big),
	\qquad
	\hat c_i=\frac{\mathrm{relu}\big(-\,g^{f}_i\,g^{r}_i\big)}{\max_j \mathrm{relu}\big(-\,g^{f}_j\,g^{r}_j\big)},
\end{equation}
که در آن $g^{f}$ و $g^{r}$ شیب فراموشی و نگه‌داری روی لایه‌ی تطبیق مشترک‌اند و $\hat c_i$ انرژی تعارض مقیاس‌بندی‌شده است. مقدار $\gamma=0$ به‌روزرسانی کامل را بازمی‌گرداند، پس حالت کنترل درون خود این بررسی قرار دارد و این آزمایش به هیچ جدول دیگری تکیه نمی‌کند.

\input{tables/res_conflict_gamma}

جدول \ref{tab:res-conflict-gamma} هم وضعیت پایانی و هم آمار خود نقاب را کنار هم می‌دهد؛ این دو باید با هم خوانده شوند، چون معنای نبود تغییر بسته به آنکه نقاب تغییر کرده باشد یا نه کاملا متفاوت است.

\paragraph{وضعیت پایانی تغییر نمی‌کند.} در سراسر بررسی، دامنه‌ی تغییر دقت نهایی روی \lr{CIFAR-10} صفر و روی \lr{CIFAR-100} تنها $0.0006$ است؛ بیشترین افت و دقت هدف نیز در همین حد جابه‌جا می‌شوند. با سه بذر، هیچ‌کدام از این دامنه‌ها از نوسان ناشی از بذر قابل‌تفکیک نیستند.

\paragraph{اما نقاب واقعا تغییر می‌کند.} این نکته، تفاوت میان یک نتیجه‌ی منفی معنادار و یک اشکال پیاده‌سازی است. در $\gamma=0$ هر $4916$ درایه‌ی زیرمجموعه‌ی فراموشی به‌روزرسانی کامل می‌گیرند، و از $\gamma=0.25$ به بعد حدود یک‌سوم آن‌ها به دسته‌ی «نرم» منتقل می‌شوند. اطلاعات تعارض هم وجود دارد و ساختگی نیست: میان $40$ تا $45$ درصد درایه‌ها انرژی تعارض ناصفر دارند. پس نه نقاب ثابت مانده و نه اطلاعات تعارض غایب است.

\paragraph{تبیین: تمرکز در بخش باریکی از توزیع.} توضیح در توزیع $\hat c$ نهفته است. این کمیت بر بیشینه تقسیم می‌شود، پس $\max_i\hat c_i=1$ است؛ اما \emph{میانگین} آن تنها $0.00106$ روی \lr{CIFAR-10} و حدود $0.004$ روی \lr{CIFAR-100} است، یعنی سه مرتبه‌ی بزرگی کمتر از بیشینه. انرژی تعارض بنابراین در بخش بسیار باریکی از توزیع متمرکز است و اکثریت قاطع درایه‌ها $\hat c_i\approx0$ دارند. با ضریب \emph{خطی} رابطه‌ی \eqref{eq:continuous-mask}، چنین درایه‌ای حتی در بزرگ‌ترین $\gamma$ آزموده‌شده ضریبی نزدیک به یک می‌گیرد. سهم درایه‌هایی که ضریبشان واقعا زیر یک‌دوم می‌افتد، در $\gamma=2$ هنوز کمتر از یک درصد است. برچسب «نرم» در جدول بنابراین گمراه‌کننده است: آن درایه‌ها از نظر تعریف نرم‌اند اما از نظر عددی تقریبا به‌طور کامل به‌روز می‌شوند.

\paragraph{نتیجه.} تبیین درشت‌دانگی رد می‌شود. ریزکردن افراز به یک ضریب پیوسته‌ی هر-درایه، رفتار آغاز تا پایان را تغییر نمی‌دهد، و دلیلش نیز مشخص است: با این صورت‌بندی، $\gamma$ باید دو تا سه مرتبه‌ی بزرگی بزرگ‌تر باشد تا درایه‌ی نوعی را به‌طور محسوس تضعیف کند. این یافته نتیجه‌ی منفی بخش پیش را تقویت می‌کند و هم‌زمان یک مسیر مشخص برای کارهای آینده باز می‌گذارد: نقابی که به‌جای \emph{مقدار} $\hat c$ بر \emph{رتبه}‌ی آن بنا شود، از این مشکل مقیاس مصون است. همچنین این بررسی همان سنجه‌ی جهت‌داری را که در بخش \ref{subsec:res-gradnorm} غایب بود فراهم می‌کند: نسبت هنجار شیب تنها بزرگی را می‌سنجید، در حالی که $\hat c$ از ضرب داخلی درایه‌ای ساخته شده و جهت را در بر دارد.

روی \lr{TinyImageNet}، دشوارترین مجموعه‌داده‌ی ارزیابی این پایان‌نامه با $20$ وظیفه، \lr{EPALL} دقت بیشتری دارد ($0.4533$ در برابر $0.3365$). بااین‌حال، بیشترین افت هر دو نسخه‌ی کامل همچنان زیاد است ($0.0840$ و $0.0853$). این نتیجه نشان می‌دهد که در محیط دارای بیست وظیفه و سه دوره‌ی آموزش، ظرفیت شبکه برای همه‌ی وظیفه‌ها کافی نیست و مشکل فقط از هم‌پوشانی ناشی نمی‌شود. بازه‌های میان بذرها هم‌پوشانی دارند؛ بنابراین نتایج سه بذر این جدول فقط روندی توصیفی را نشان می‌دهند و رتبه‌بندی روش‌ها را تأیید نمی‌کنند.

\subsection{آماره‌ی تفکیک‌پذیری حمله‌ی آستانه}
\label{subsec:res-epsilon}

نبود تفاوت در بخش پیش تنها می‌گوید یک آزمون با توان تشخیص پایین چیزی پیدا نکرد. سنجه‌ی حساس‌تر آن است که به‌جای \lr{AUC} میانگین، بدترین نقطه‌ی کاری همان حمله را ببینیم. برای این کار عضو را کلاس مثبت می‌گیریم و روی شبکه‌ای از آستانه‌ها، کران پایین یک‌طرفه‌ی ۹۵ درصد کلاپر-پیرسون برای نرخ مثبت درست و کران بالای همان‌گونه برای نرخ مثبت نادرست را می‌سازیم؛ آنگاه
\begin{equation}
	\label{eq:eps-hat}
	S=\max_{\text{آستانه}}\ \log\frac{\mathrm{TPR}_{\text{کران پایین}}}{\mathrm{FPR}_{\text{کران بالا}}},
\end{equation}
که در صفر بریده می‌شود. \textbf{این کمیت، پیراسنجه‌ی حریم خصوصی تفاضلی یا کران پایین آن نیست.} در بازخورد بررسی، ارزیابی تجربی حریم خصوصی و گزارش $(\epsilon,\delta)$ درخواست شده بود. آزمایش انجام شد و نتیجه‌ی آن در همین جدول آمده است، اما نمی‌توان مقدار به‌دست‌آمده را به‌صورت $(\epsilon,\delta)$ تفسیر کرد. سه شرط لازم برقرار نیست: آستانه روی همان داده‌ای انتخاب می‌شود که آماره روی آن سنجیده می‌شود؛ ساختار مجموعه‌داده‌های همسایه وجود ندارد؛ و شمول تصادفی و پیراسنجه‌ی $\delta$ تعریف نشده‌اند. بنابراین، $S$ فقط آماره‌ی تفکیک‌پذیری همین حمله‌ی خاص است. ارزیابی رسمی به روش \cite{steinke2023privacy} انجام نشده و در بخش \ref{sec:concl-future} به‌عنوان کار آینده پیشنهاد می‌شود. جدول \ref{tab:res-epsilon} نتیجه را نشان می‌دهد.

\input{tables/res_epsilon}

این جدول نتیجه‌ای نامطلوب برای روش پیشنهادی نشان می‌دهد. پیش از درخواست، $S$ برای \lr{EPALL} روی هر دو مجموعه‌داده صفر است؛ پس از درخواست، میانگین آن به $0.0993$ و $0.0995$ و بیشینه‌ی آن به $0.4555$ و $0.5544$ می‌رسد. بنابراین، عملیات حذف در این ارزیابی نمونه‌های وظیفه‌ی هدف را \emph{تفکیک‌پذیرتر} کرده است. یک توضیح احتمالی آن است که بازنشانی و ترمیم، رفتار مدل روی نمونه‌های حذف‌شده را از الگوی نمونه‌های دیده‌نشده دور می‌کند و این تفاوت برای حمله قابل استفاده می‌شود. نزدیک‌بودن دقت هدف به حدس تصادفی با این یافته ناسازگار نیست؛ مدل ممکن است پاسخ درست ندهد، اما همچنان روی نمونه‌های عضو و غیرعضو به شکل متفاوتی اشتباه کند.

\lr{PALL} اصلی، که با همان بذرها، وظیفه‌های حذف‌شده و روند حمله سنجیده شده است، این افزایش را نشان نمی‌دهد. میانگین $S$ آن پس از درخواست روی \lr{CIFAR-10} برابر $0.0013$ و روی \lr{CIFAR-100} صفر است. بنابراین، افزایش تفکیک‌پذیری به بخش محافظت و ترمیم لنگردار \lr{EPALL} مربوط است و باید به‌عنوان هزینه‌ای در کنار بهبود نگه‌داری در نظر گرفته شود. \lr{CLPU} پس از حذف کامل شبکه‌ی وظیفه‌ی هدف در هر دو مجموعه‌داده به $S=0$ می‌رسد. \lr{PALL-Adapter} و \lr{LoRA} نیز با بدنه‌ی ثابت مقادیر بسیار کوچک‌تری دارند که با دامنه‌ی محدود به‌روزرسانی آن‌ها سازگار است. مقدار بزرگ $S$ تفکیک‌پذیری را نشان می‌دهد، اما مقدار کوچک تضمینی برای حذف نیست؛ زیرا ممکن است حمله‌ی به‌کاررفته توان کافی نداشته باشد. ازاین‌رو، این جدول می‌تواند شاهدی علیه حذف موفق فراهم کند، اما به‌تنهایی حذف را تأیید نمی‌کند.

\subsection{نسبت هنجار شیب}
\label{subsec:res-gradnorm}

معیار آخر این بخش، شدت \emph{تداخل} را در لحظه‌ی درخواست می‌سنجد و نه کیفیت فراموشی را: نسبت هنجار $\ell_2$ شیب وظیفه‌ی هدف به شیب وظایف نگه‌داری‌شده، هر دو روی زیرشبکه‌ی مشترک و پیش از بازنشانی. جدول \ref{tab:res-gradnorm} این نسبت را می‌دهد.

\input{tables/res_gradnorm}

مقادیر در بازه‌ی $0.67$ تا $0.80$ قرار می‌گیرند، یعنی دو شیب \emph{هم‌مرتبه} هستند: شیب فراموشی نه آن‌قدر بزرگ است که شیب نگه‌داری را بی‌اثر کند و نه آن‌قدر کوچک که نادیده گرفته شود.

\textbf{اما این معیار تعارض را نمی‌سنجد.} چون $\|g_f\|_2/\|g_r\|_2$ تنها نسبت \emph{بزرگی} است و هیچ اطلاعاتی از جهت ندارد، دو شیب می‌توانند این نسبت را برابر یک داشته باشند و در عین حال کاملا هم‌راستا یا کاملا متضاد باشند. بنابراین این جدول تنها نشان می‌دهد دو شیب در ناحیه‌ی مشترک \emph{قابل مقایسه‌اند}، و ادعای «رقابت» یا «تعارض» از آن بیرون نمی‌آید. سنجش تعارض به معیارهای جهت‌دار نیاز دارد — کسینوس زاویه، ضرب داخلی، یا درصد درایه‌هایی که علامت مخالف دارند — که هیچ‌کدام در این کار محاسبه نشده‌اند و در بخش \ref{sec:concl-future} به‌عنوان کار آینده ثبت می‌شوند. تفاوت میان \lr{EPALL} و \lr{PALL} اصلی در این معیار کوچک و در حد انحراف معیار است، که انتظار هم همین بود: این نسبت پیش از اعمال محافظت اندازه‌گیری می‌شود و ویژگی \emph{مسئله} است و نه ویژگی روش.

\subsection{ماندگاری سرکوب}
\label{subsec:res-persistence}

پژوهش‌هایی مانند \lr{UnCLe} \cite{adhikari2026uncle} و \lr{CATA} \cite{lin2026cata} نشان می‌دهند که دقت بلافاصله پس از درخواست کافی نیست. جدول \ref{tab:res-persistence} وظیفه‌ی فراموش‌شده را از لحظه‌ی حذف تا پایان دنباله‌ی درخواست‌ها دنبال می‌کند. معیار این ارزیابی \emph{فاصله تا حدس تصادفی} $|A_u-c|$ است، نه دقت خام. هدف حذف، نزدیک‌کردن عملکرد به حدس تصادفی است؛ بنابراین فاصله‌گرفتن در هر دو جهت نامطلوب است.

در دو بذر این ارزیابی ($0$ و $1$)، پس از نخستین حذف درخواست آموزشی نیز وجود دارد. برای بذر $0$، الگوی \lr{CIFAR-10} برابر \lr{T T T T F T F F} و الگوی \lr{CIFAR-100} برابر \lr{T$\times$7 F T T F T F} است. این الگو در همه‌ی بذرها یکسان نیست؛ برای نمونه، در بذر $5$ مجموعه‌داده‌ی \lr{CIFAR-100} هر سه حذف در پایان دنباله قرار دارند (\lr{T$\times$10 F F F}) و پس از نخستین حذف هیچ آموزشی انجام نمی‌شود. رانش وظیفه‌ی فراموش‌شده به دو بخش تقسیم می‌شود: تغییر ناشی از درخواست‌های آموزشی میان دو حذف و تغییر ناشی از درخواست‌های حذف بعدی. این دو بخش با استفاده از شناسه‌ی رویداد و مقادیر پیش و پس از هر حذف محاسبه می‌شوند؛ اگر میان دو حذف آموزشی انجام نشده باشد، سهم آموزش صفر است.

\input{tables/res_persistence}

بدترین بازگشت مشاهده‌شده روی \lr{CIFAR-10} برابر $0.0490$ برای \lr{PALL} و $0.0370$ برای \lr{EPALL} و $0.0060$ برای \lr{PALL-Adapter} است؛ روی \lr{CIFAR-100} برابر $0.0500$ برای \lr{PALL-Adapter} و $0.0100$ برای \lr{PALL} و \lr{EPALL}. \lr{LoRA} و \lr{CLPU} بازگشت دقیقا صفر دارند، چون پیراسنجه‌های وظیفه‌ی حذف‌شده در آن‌ها از پیراسنجه‌های وظایف بعدی جدا هستند و هیچ درخواست بعدی آن‌ها را لمس نمی‌کند.

تفکیک سهم‌ها نتیجه‌ی غیرمنتظره‌ای می‌دهد. روی \lr{CIFAR-100}، سهم درخواست‌های آموزشی برای هر دو نسخه‌ی \lr{PALL} \emph{منفی} است ($-0.0090$ و $-0.0120$)؛ یعنی یادگیری وظایف تازه، وظیفه‌ی فراموش‌شده را به شانس نزدیک‌تر می‌کند نه دورتر. روی \lr{CIFAR-10} همین سهم اندکی مثبت است ($+0.0019$ و $+0.0063$) اما بخش عمده‌ی رانش \lr{PALL} از درخواست‌های \emph{حذف} بعدی می‌آید ($+0.0111$)، در حالی که همین سهم برای \lr{EPALL} منفی است ($-0.0053$). تفسیر این تفاوت آن است که حفاظت رتبه‌بندی‌شده، علاوه بر مهار آسیب به وظایف نگه‌داری‌شده، مانع از آن می‌شود که حذف‌های بعدی وظیفه‌ی پیش‌تر حذف‌شده را دوباره از سطح تصادفی دور کنند. با این حال هیچ‌کدام از این تفاوت‌ها با دو بذر و چهار وظیفه‌ی دارای افق پیگیری، آزمون معناداری را برنمی‌تابد؛ آن‌ها را باید مشاهده‌ی توصیفی خواند. جمع‌بندی این است که سرکوب ماندگار است اما کاملا ثابت نمی‌ماند، و علت اصلی تغییر آن درخواست‌های حذف بعدی است نه یادگیری تازه. شکل \ref{fig:res-persistence} مسیرها را نشان می‌دهد.

\section{مطالعات حساسیت}
\label{sec:res-sensitivity}

\subsection{انتخاب لنگر}
\label{subsec:res-anchor}

بخش \ref{subsec:prop-repair} اشاره کرد که لنگرزدن به $\theta^{\mathrm{old}}$ یک ایراد مفهومی دارد: آن وزن‌ها خودشان اطلاعاتی از وظیفه‌ی فراموش‌شونده حمل می‌کنند. جدول \ref{tab:res-anchor} جایگزین $\theta^{\mathrm{reinit}}$ را روی هشت بذر هم‌تراز از طرح آزمایشی استاندارد می‌سنجد. چون \lr{\texttt{protect\_anchor}} فقط روی \lr{EPALL} اثر دارد، دو حالت آزمایشی تنها در همین یک تنظیم تفاوت دارند و بذرها جفت می‌شوند؛ حالت آزمایشی \lr{\texttt{reinit}} زیر برچسب آزمایش مستقل \lr{\texttt{*\_standard\_reinit\_v1}} اجرا شده تا جدول‌های اصلی دست‌نخورده بمانند.

\input{tables/res_anchor}

نتیجه‌ی حاصل از هشت بذر با مطالعه‌ی پیشین با دو بذر هم‌خوان نیست و جهت آن روی دو مجموعه‌داده یکسان نیست. روی \lr{CIFAR-100} برتری با $\theta^{\mathrm{old}}$ است: فراموشی در هفت جفت از هشت جفت به سود آن است ($d_z = 1.06$)، بیشترین افت در شش جفت ($d_z = 1.04$، میانگین اختلاف $0.0089$) و دقت نهایی در شش جفت. روی \lr{CIFAR-10} تفاوت‌ها در حد نوسان بذرهاست و اندکی به سود $\theta^{\mathrm{reinit}}$ می‌چرخد، اما هیچ معیاری بیش از پنج جفت از هشت جفت را نمی‌گیرد و بزرگی اختلاف دقت $0.0013$ است. تنها معیاری که در هر دو مجموعه‌داده به سود $\theta^{\mathrm{reinit}}$ می‌ماند، فاصله‌ی دقت هدف تا شانس است و آن هم با اختلاف میانگین $0.0030$ روی \lr{CIFAR-100}. مهم‌تر از همه، این خانواده ده آزمون دارد و پس از اصلاح \lr{Holm} \emph{هیچ} معیاری در هیچ جهتی در سطح $0.05$ باقی نمی‌ماند؛ کوچک‌ترین $p$ خام برابر $0.035$ است که پس از اصلاح به $0.35$ می‌رسد.

پیش‌فرض پیاده‌سازی $\theta^{\mathrm{old}}$ است و همه‌ی جدول‌های این پایان‌نامه با همان تولید شده‌اند. سه دلیل پشت این انتخاب است. نخست، لنگر تنها روی $S_{\mathrm{crit}} \subseteq M_f \cap M_r$ عمل می‌کند و فراموشی از راه بازنشانی ناحیه‌ی انحصاری $M_f \setminus M_r$ رخ می‌دهد؛ پس لنگر یک اهرم \emph{نگه‌داری} است نه اهرم فراموشی، و باید عمدتا با معیارهای نگه‌داری داوری شود — جایی که روی \lr{CIFAR-100} برتری با $\theta^{\mathrm{old}}$ است و روی \lr{CIFAR-10} تفاوتی نیست. دوم، $\theta^{\mathrm{reinit}}$ آن استقلال از داده‌ای را که وعده می‌دهد در عمل فراهم نمی‌کند: شاخه‌های بازنشانی ارثی همچنان به پشتیبان‌های تنک وابسته به داده‌ی قدیمی تکیه دارند، پس مسیر نشت از راه دیگری باز می‌ماند و محدودیت دامنه‌ی ادعای حذف (بخش \ref{sec:concl-limitations}) همچنان برقرار است. سوم، $\theta^{\mathrm{old}}$ در زمان درخواست نیز سریع‌تر است ($3.87$ ثانیه روی \lr{CIFAR-100} و $0.47$ ثانیه روی \lr{CIFAR-10} به‌طور میانگین). ایراد مفهومی $\theta^{\mathrm{old}}$ سر جای خود باقی است و در بخش \ref{sec:concl-limitations} به‌عنوان محدودیت ثبت می‌شود، اما شاهد حاصل از هشت بذر آن را به توصیه‌ی عملی تبدیل نمی‌کند.

در بازخورد بررسی، $\theta^{\mathrm{reinit}}$ به‌عنوان مرجع جمله‌ی لنگر پیشنهاد شده بود؛ زیرا توزیع مقدارگذاری اولیه از داده‌ها مستقل است و اطلاعات وظیفه‌ی حذف‌شونده را دوباره وارد نمی‌کند. این استدلال برای خود جمله‌ی لنگر درست است، اما در سطح کل سامانه کافی نیست. مسیر بازنشانی \lr{PALL} همچنان از پشتیبان‌های تنک وابسته به داده‌های پیشین استفاده می‌کند؛ بنابراین، مستقل‌کردن لنگر فقط یکی از مسیرهای وابستگی را حذف می‌کند. افزون بر این، تضمین حذف دقیق فقط زمانی برقرار است که حذف روی پیراسنجه‌های غیرمشترک نیز دقیق باشد. ارزیابی مرجع بازآموزی در بخش \ref{subsec:res-retrain} این شرط را تأیید نمی‌کند؛ توافق روی وظیفه‌ی هدف میان $0.00$ و $0.98$ تغییر می‌کند. مقایسه‌ی هشت بذر نیز هیچ برتری تجربی برای $\theta^{\mathrm{reinit}}$ نشان نداد. به همین دلیل، پیشنهاد بررسی شد اما در روش نهایی پذیرفته نشد.

\section{تهدیدهای اعتبار}
\label{sec:res-threats}

\paragraph{اعتبار درونی.} خطر اصلی، انتساب بهبود به مؤلفه‌ی نادرست است. کاهش این خطر با سه سازوکار انجام شده است: جفت‌شدگی دنباله‌ها بر اساس بذر، کنترل‌های مؤلفه‌ای که هر بار فقط یک جزء را عوض می‌کنند، و هم‌ترازی مولد اعداد تصادفی در حالت‌های لایه‌ی تطبیق که باعث شد بی‌اثربودن نقاب نرم از نوسان تصادفی قابل تفکیک باشد.

\paragraph{اعتبار آماری.} کف $p$ قابل‌دستیابی برای هر آزمون جداگانه است و به تعداد جفت‌های بدون تساوی بستگی دارد؛ برای سطرهای بدون تساوی \lr{CIFAR-100} برابر $0.003906$ و برای دقت نهایی \lr{CIFAR-10} برابر $0.007812$ است. شش آزمون هم‌زمان با اصلاح \lr{Holm} تصحیح شده‌اند و چهار مورد از آن‌ها در سطح $0.05$ باقی می‌مانند. تحلیل‌های مبتنی بر دو و سه بذر صریحا توصیفی برچسب خورده‌اند و در هیچ ادعای معناداری وارد نشده‌اند. بازه‌های بازنمونه‌گیری نیز توصیفی‌اند و آزمون فرض نیستند.

\paragraph{ناپایداری نتیجه با افزودن بذر.} یک هشدار روش‌شناختی مشخص از همین کار بیرون آمد: افزودن سه بذر به کنترل‌های مؤلفه‌ای، جدایی جمله‌ی لنگر روی \lr{CIFAR-100} را از میان برد (بخش \ref{subsec:res-components}). این نشان می‌دهد بازه‌های بازنمونه‌گیری نمونه‌ی کوچک تا چه اندازه می‌توانند شکننده باشند و چرا نباید انتساب سازوکار را تنها بر پایه‌ی کمترین تعداد بذری گزارش کرد که نتیجه در آن پایدار به نظر می‌رسد.

\paragraph{اعتبار بیرونی.} همه‌ی نتایج روی سه مجموعه‌داده‌ی تصویری با تقسیم مصنوعی وظیفه‌ها به دست آمده‌اند. تعمیم به داده‌های واقعی با مرزهای نامشخص وظیفه، داده‌های چندحالتی یا مقیاس‌های بزرگ‌تر آزموده نشده است. آزمون \lr{ViT} فقط نشانه‌ای اولیه از انتقال میان معماری‌ها فراهم می‌کند.

\paragraph{اعتبار سازه‌ای.} خطر جدی‌تر، گرفتن معیارها به‌جای مفهوم است. سه نمونه‌ی مشخص: $A_u$ نزدیک به سطح تصادفی معیار حذف نیست، $\Rupd$ معیار کارایی نیست، و \lr{AUC} نزدیک به $0.5$ در \lr{MIA} معیار حریم خصوصی نیست. تمام متن این فصل بر همین تفکیک‌ها اصرار کرده است.

\paragraph{اعتبار محاسباتی.} زمان درخواست روی یک پیکربندی سخت‌افزاری مشخص اندازه‌گیری شده و برای سخت‌افزارهای متفاوت مقیاس‌بندی نشده است. بنابراین، فقط برای مقایسه‌ی تقریبی روش‌ها در همین محیط مناسب است.

\section{جمع‌بندی یافته‌ها}
\label{sec:res-summary}

پاسخ‌های تجربی به پرسش‌های پژوهش در پنج بند زیر جمع‌بندی می‌شوند؛ هر پاسخ فقط در محدوده‌ی طرح‌های آزمایشی و تعداد بذرهای گزارش‌شده معتبر است.

\begin{description}
	\item[\lr{RQ1} — بله، با دامنه‌ی مشخص.] \lr{EPALL} هم‌زمان $A_u$ را به سطح تصادفی می‌رساند و دقت نگه‌داری را نسبت به \lr{PALL} اصلی بهبود می‌دهد؛ این بهبود روی هر دو مجموعه‌داده پس از اصلاح \lr{Holm} باقی می‌ماند. بیشترین افت تنها روی \lr{CIFAR-100} معنادار می‌ماند.
	\item[\lr{RQ2} — دقت نزدیک، با زمان پردازش بیشتر.] دقت \lr{EPALL} روی \lr{CIFAR-10} کمتر از یک واحد درصد پایین‌تر از \lr{CLPU} است و روی \lr{CIFAR-100} در محدوده‌ی پراکندگی بذرها با آن تفاوت اندکی دارد. در مقابل، \lr{CLPU} به $1.6$ تا $2.2$ برابر حافظه‌ی بیشتر نیاز دارد و \lr{EPALL} درخواست را کندتر از \lr{PALL} اصلی پردازش می‌کند.
	\item[\lr{RQ3} — شواهد مثبت اما مشروط؛ اثر مستقل هم‌پوشانی اثبات نشده است.] آزمون موقعیت درخواست روندی توصیفی نشان می‌دهد: بیشترین افت \lr{PALL} اصلی با افزایش سطح بالا می‌رود و \lr{EPALL} تقریبا ثابت می‌ماند، اما سطوح آزمایش هم‌پوشانی واقعی را کنترل نمی‌کنند. در بررسی تنکی (بخش \ref{sec:res-overlap-direct}) هویت و موقعیت وظیفه‌ی هدف ثابت می‌مانند و هم‌پوشانی اندازه‌گیری‌شده بیش از ده برابر تغییر می‌کند. میان هم‌پوشانی و میزان بهبود، همبستگی رتبه‌ای مثبت و معناداری وجود دارد؛ اما چون تنکی ظرفیت را نیز تغییر می‌دهد، اثر مستقل هم‌پوشانی قابل تعیین نیست.
	\item[\lr{RQ4} — تنها رتبه‌بندی با بودجه‌ی ثابت.] کیفیت انتخاب به‌روشنی و روی هر دو مجموعه‌داده مؤثر است. دو جزء دیگر سهم مستقل نشان نمی‌دهند، اما کنترل‌های موجود برای رد ضرورت آن‌ها کافی نیستند. افزون بر این، رتبه‌بندی در همه‌ی معیارها برتر نیست: محافظت بدون رتبه‌بندی بیشترین افت کمتری دارد.
	\item[\lr{RQ5} — نه.] نقاب نرم در نقطه‌ی پایان مسیر لایه‌ی تطبیق اثر قابل‌اندازه‌گیری ندارد و سرکوب به بازنشانی مسیر هدف نسبت داده می‌شود.
\end{description}
